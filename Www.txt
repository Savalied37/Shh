local Rayfield = loadstring(game:HttpGet("https://sirius.menu/rayfield"))()

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local LocalPlayer = Players.LocalPlayer

local Window = Rayfield:CreateWindow({
    Name = "Admin Ban Panel",
    LoadingTitle = "Ban System",
    LoadingSubtitle = "Loading...",
    ConfigurationSaving = {
        Enabled = false
    },
    KeySystem = false
})

local Tab = Window:CreateTab("Ban Player")

local SelectedPlayer = nil
local BanDuration = "24"
local BanReason = "No reason"

local function GetPlayerNames()
    local names = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            table.insert(names, plr.Name)
        end
    end
    if #names == 0 then
        return {"No players available"}
    end
    return names
end

local function FindPlayerByName(name)
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr.Name == name then
            return plr
        end
    end
    return nil
end

local function BanPlayer(targetPlayer, duration, reason)
    if not targetPlayer or not targetPlayer.Parent then
        Rayfield:Notify({
            Title = "❌ Error",
            Content = "No player selected or player left!",
            Duration = 3
        })
        return false
    end

    -- UserID и Duration передаём как СТРОКИ
    local args = {
        [1] = "BanPlayer",
        [2] = tostring(targetPlayer.UserId),
        [3] = tostring(duration),
        [4] = tostring(reason)
    }

    local success, err = pcall(function()
        game:GetService("ReplicatedStorage").Admin:FireServer(
            args[1],
            args[2],
            args[3],
            args[4]
        )
    end)

    if success then
        print("✅ BAN SENT")
        print("Action:", args[1])
        print("UserID:", args[2])
        print("Duration:", args[3])
        print("Reason:", args[4])
        
        Rayfield:Notify({
            Title = "✅ BANNED",
            Content = string.format("%s\nID: %s\nDuration: %sh\nReason: %s",
                targetPlayer.Name,
                args[2],
                args[3],
                args[4]
            ),
            Duration = 6,
            Image = 4483362458
        })
        return true
    else
        print("❌ BAN FAILED:", err)
        Rayfield:Notify({
            Title = "❌ Ban Failed",
            Content = tostring(err),
            Duration = 5
        })
        return false
    end
end

-- Dropdown
local PlayerDropdown = Tab:CreateDropdown({
    Name = "Select Player",
    Options = GetPlayerNames(),
    CurrentOption = GetPlayerNames()[1] or "No players available",
    MultipleOptions = false,
    Flag = "PlayerSelect",
    Callback = function(option)
        local selectedName = nil
        if type(option) == "table" then
            selectedName = option[1]
        elseif type(option) == "string" then
            selectedName = option
        end

        if selectedName and selectedName ~= "No players available" then
            local plr = FindPlayerByName(selectedName)
            if plr then
                SelectedPlayer = plr
                print("Selected:", plr.Name, "| ID:", plr.UserId)
                Rayfield:Notify({
                    Title = "✅ Selected",
                    Content = plr.Name .. "\nID: " .. tostring(plr.UserId),
                    Duration = 3
                })
            end
        else
            SelectedPlayer = nil
        end
    end
})

Tab:CreateButton({
    Name = "🔄 Refresh List",
    Callback = function()
        SelectedPlayer = nil
        local newList = GetPlayerNames()
        PlayerDropdown:Refresh(newList, true)
        Rayfield:Notify({
            Title = "Refreshed",
            Content = "Found " .. #newList .. " player(s)",
            Duration = 2
        })
    end
})

Tab:CreateSection("Ban Settings")

Tab:CreateInput({
    Name = "Duration (hours)",
    PlaceholderText = "24",
    RemoveTextAfterFocusLost = false,
    Callback = function(Text)
        if Text and Text ~= "" and tonumber(Text) then
            BanDuration = Text -- Сохраняем как строку
            Rayfield:Notify({
                Title = "Duration Set",
                Content = Text .. " hours",
                Duration = 2
            })
        else
            Rayfield:Notify({
                Title = "Error",
                Content = "Enter a valid number",
                Duration = 2
            })
        end
    end
})

Tab:CreateInput({
    Name = "Ban Reason",
    PlaceholderText = "Enter reason",
    RemoveTextAfterFocusLost = false,
    Callback = function(Text)
        if Text and Text ~= "" then
            BanReason = Text
            Rayfield:Notify({
                Title = "Reason Set",
                Content = Text,
                Duration = 2
            })
        end
    end
})

Tab:CreateSection("Execute")

Tab:CreateButton({
    Name = "🔨 BAN PLAYER",
    Callback = function()
        BanPlayer(SelectedPlayer, BanDuration, BanReason)
    end
})

Tab:CreateSection("Quick Ban")

local function QuickBan(hours, reason)
    BanPlayer(SelectedPlayer, tostring(hours), reason)
end

Tab:CreateButton({
    Name = "⚡ 1h - Spam",
    Callback = function() QuickBan("1", "Spamming") end
})

Tab:CreateButton({
    Name = "⚡ 6h - Toxicity",
    Callback = function() QuickBan("6", "Toxic behavior") end
})

Tab:CreateButton({
    Name = "⚡ 24h - Exploiting",
    Callback = function() QuickBan("24", "Exploiting") end
})

Tab:CreateButton({
    Name = "⚡ 168h - 7 Days",
    Callback = function() QuickBan("168", "Severe violation") end
})

Tab:CreateButton({
    Name = "⚡ 720h - 30 Days",
    Callback = function() QuickBan("720", "Permanent-like ban") end
})

Tab:CreateSection("Duration Presets")

local function SetDuration(hours)
    BanDuration = tostring(hours)
    Rayfield:Notify({
        Title = "Duration Set",
        Content = hours .. " hours",
        Duration = 2
    })
end

Tab:CreateButton({Name = "⏱️ 1h",   Callback = function() SetDuration("1") end})
Tab:CreateButton({Name = "⏱️ 6h",   Callback = function() SetDuration("6") end})
Tab:CreateButton({Name = "⏱️ 12h",  Callback = function() SetDuration("12") end})
Tab:CreateButton({Name = "⏱️ 24h",  Callback = function() SetDuration("24") end})
Tab:CreateButton({Name = "⏱️ 72h",  Callback = function() SetDuration("72") end})
Tab:CreateButton({Name = "⏱️ 168h", Callback = function() SetDuration("168") end})
Tab:CreateButton({Name = "⏱️ 720h", Callback = function() SetDuration("720") end})

-- Players Tab
local InfoTab = Window:CreateTab("Players")

InfoTab:CreateButton({
    Name = "📋 List All Players",
    Callback = function()
        local list = Players:GetPlayers()
        print("=== PLAYERS ===")
        for i, plr in ipairs(list) do
            if plr ~= LocalPlayer then
                local info = i .. ". " .. plr.Name .. " | ID: " .. tostring(plr.UserId)
                print(info)
                Rayfield:Notify({
                    Title = plr.Name,
                    Content = "ID: " .. tostring(plr.UserId),
                    Duration = 3
                })
                task.wait(0.4)
            end
        end
        print("===============")
    end
})

InfoTab:CreateButton({
    Name = "🎯 Selected Player Info",
    Callback = function()
        if SelectedPlayer and SelectedPlayer.Parent then
            local info = string.format(
                "Name: %s\nUserID: %s\nDisplay: %s\nAge: %d days",
                SelectedPlayer.Name,
                tostring(SelectedPlayer.UserId),
                SelectedPlayer.DisplayName,
                SelectedPlayer.AccountAge
            )
            print(info)
            Rayfield:Notify({
                Title = SelectedPlayer.Name,
                Content = info,
                Duration = 5
            })
        else
            Rayfield:Notify({
                Title = "No Selection",
                Content = "Select a player first",
                Duration = 3
            })
        end
    end
})

-- Auto refresh
Players.PlayerAdded:Connect(function(player)
    task.wait(0.5)
    pcall(function()
        PlayerDropdown:Refresh(GetPlayerNames(), true)
        Rayfield:Notify({
            Title = "Joined",
            Content = player.Name,
            Duration = 2
        })
    end)
end)

Players.PlayerRemoving:Connect(function(player)
    task.wait(0.5)
    pcall(function()
        if SelectedPlayer == player then
            SelectedPlayer = nil
            Rayfield:Notify({
                Title = "Player Left",
                Content = player.Name .. " left - selection cleared",
                Duration = 3
            })
        end
        PlayerDropdown:Refresh(GetPlayerNames(), true)
    end)
end)

print("=== Admin Ban Panel Loaded ===")
