local Rayfield = loadstring(game:HttpGet("https://sirius.menu/rayfield"))()

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local LocalPlayer = Players.LocalPlayer

local Window = Rayfield:CreateWindow({
    Name = "Admin Ban Panel",
    LoadingTitle = "Ban System",
    LoadingSubtitle = "Loading...",
    ConfigurationSaving = {
        Enabled = false
    },
    KeySystem = false
})

local Tab = Window:CreateTab("Ban Player")

local SelectedPlayer = nil
local BanDuration = 24
local BanReason = "No reason"

local function GetPlayerNames()
    local names = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            table.insert(names, plr.Name)
        end
    end
    if #names == 0 then
        table.insert(names, "No players")
    end
    return names
end

-- Dropdown
local PlayerDropdown = Tab:CreateDropdown({
    Name = "Select Player",
    Options = GetPlayerNames(),
    CurrentOption = {"No players"},
    MultipleOptions = false,
    Flag = "PlayerSelect",
    Callback = function(Option)
        local name = Option
        if name and name ~= "No players" then
            local plr = Players:FindFirstChild(name)
            if plr then
                SelectedPlayer = plr
                Rayfield:Notify({
                    Title = "Selected",
                    Content = plr.Name .. " | ID: " .. plr.UserId,
                    Duration = 3
                })
            end
        end
    end
})

-- Refresh button
Tab:CreateButton({
    Name = "Refresh Player List",
    Callback = function()
        local newList = GetPlayerNames()
        PlayerDropdown:Refresh(newList)
        Rayfield:Notify({
            Title = "Refreshed",
            Content = "Player list updated",
            Duration = 2
        })
    end
})

Tab:CreateSection("Ban Settings")

-- Duration input
Tab:CreateInput({
    Name = "Ban Duration (hours)",
    PlaceholderText = "24",
    RemoveTextAfterFocusLost = false,
    Callback = function(Text)
        local num = tonumber(Text)
        if num and num > 0 then
            BanDuration = num
            Rayfield:Notify({
                Title = "Duration Set",
                Content = num .. " hours",
                Duration = 2
            })
        else
            Rayfield:Notify({
                Title = "Error",
                Content = "Enter a valid number",
                Duration = 2
            })
        end
    end
})

-- Reason input
Tab:CreateInput({
    Name = "Ban Reason",
    PlaceholderText = "Enter reason here",
    RemoveTextAfterFocusLost = false,
    Callback = function(Text)
        if Text and Text ~= "" then
            BanReason = Text
            Rayfield:Notify({
                Title = "Reason Set",
                Content = BanReason,
                Duration = 2
            })
        end
    end
})

Tab:CreateSection("Execute Ban")

-- BAN BUTTON
Tab:CreateButton({
    Name = "🔨 BAN PLAYER",
    Callback = function()
        if SelectedPlayer == nil then
            Rayfield:Notify({
                Title = "Error",
                Content = "Select a player first!",
                Duration = 3
            })
            return
        end

        local success, err = pcall(function()
            local AdminRemote = ReplicatedStorage:WaitForChild("Admin", 5)
            if AdminRemote then
                AdminRemote:FireServer("BanPlayer", SelectedPlayer.UserId, BanDuration, BanReason)
            else
                error("Admin remote not found")
            end
        end)

        if success then
            Rayfield:Notify({
                Title = "✅ Banned!",
                Content = SelectedPlayer.Name .. " for " .. BanDuration .. "h - " .. BanReason,
                Duration = 5
            })
        else
            Rayfield:Notify({
                Title = "❌ Error",
                Content = tostring(err),
                Duration = 5
            })
        end
    end
})

Tab:CreateSection("Quick Duration Presets")

-- Quick presets
Tab:CreateButton({
    Name = "⏱️ 1 Hour",
    Callback = function()
        BanDuration = 1
        Rayfield:Notify({Title = "Duration", Content = "Set to 1 hour", Duration = 2})
    end
})

Tab:CreateButton({
    Name = "⏱️ 24 Hours",
    Callback = function()
        BanDuration = 24
        Rayfield:Notify({Title = "Duration", Content = "Set to 24 hours", Duration = 2})
    end
})

Tab:CreateButton({
    Name = "⏱️ 7 Days (168h)",
    Callback = function()
        BanDuration = 168
        Rayfield:Notify({Title = "Duration", Content = "Set to 7 days", Duration = 2})
    end
})

Tab:CreateButton({
    Name = "⏱️ 30 Days (720h)",
    Callback = function()
        BanDuration = 720
        Rayfield:Notify({Title = "Duration", Content = "Set to 30 days", Duration = 2})
    end
})

Tab:CreateButton({
    Name = "⏱️ Permanent (999999h)",
    Callback = function()
        BanDuration = 999999
        Rayfield:Notify({Title = "Duration", Content = "Set to PERMANENT", Duration = 2})
    end
})

-- Info tab
local InfoTab = Window:CreateTab("Player Info")

InfoTab:CreateButton({
    Name = "📋 Show All Players",
    Callback = function()
        local count = 0
        for _, plr in ipairs(Players:GetPlayers()) do
            count = count + 1
            print(plr.Name .. " | " .. plr.UserId)
            Rayfield:Notify({
                Title = "Player #" .. count,
                Content = plr.Name .. " - ID: " .. plr.UserId,
                Duration = 3
            })
            task.wait(0.5)
        end
    end
})

InfoTab:CreateButton({
    Name = "ℹ️ Selected Player Info",
    Callback = function()
        if SelectedPlayer then
            Rayfield:Notify({
                Title = "Current Selection",
                Content = SelectedPlayer.Name .. "\nID: " .. SelectedPlayer.UserId .. "\nAccount Age: " .. SelectedPlayer.AccountAge .. " days",
                Duration = 5
            })
        else
            Rayfield:Notify({
                Title = "No Selection",
                Content = "Please select a player first",
                Duration = 3
            })
        end
    end
})

-- Auto refresh player list
Players.PlayerAdded:Connect(function()
    task.wait(1)
    pcall(function()
        PlayerDropdown:Refresh(GetPlayerNames())
    end)
end)

Players.PlayerRemoving:Connect(function()
    task.wait(1)
    pcall(function()
        PlayerDropdown:Refresh(GetPlayerNames())
        if SelectedPlayer and not SelectedPlayer.Parent then
            SelectedPlayer = nil
        end
    end)
end)

Rayfield:LoadConfiguration()
