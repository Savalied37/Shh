local function sendWebhook(username, placeName, action)
    local payload = {
        username = "TDR Hub Logger",
        embeds = {{
            title = "The Depths of Reality - Action Log",
            description = "User: " .. username .. "\nAction: " .. action,
            fields = {
                {name = "Username", value = username, inline = true},
                {name = "Place", value = placeName, inline = false},
                {name = "Timestamp", value = os.date("%Y-%m-%d %H:%M:%S"), inline = false}
            },
            color = 3066993
        }}
    }
    
    local httpRequest = syn and syn.request or http and http.request or http_request or request
    if not httpRequest then return end
    
    pcall(function()
        httpRequest({
            Url = WEBHOOK_URL,
            Method = "POST",
            Headers = {["Content-Type"] = "application/json"},
            Body = HttpService:JSONEncode(payload)
        })
    end)
end

local function createToggleButton(parent, name, yPos, width, callback)
    local btn = Instance.new("TextButton")
    btn.Name = name
    btn.Size = UDim2.new(width, 0, 0, 38)
    btn.Position = UDim2.new(0, 0, 0, yPos)
    btn.BackgroundColor3 = Color3.fromRGB(45, 45, 65)
    btn.TextColor3 = Color3.fromRGB(180, 180, 200)
    btn.TextSize = 11
    btn.Text = name .. " [OFF]"
    btn.Font = Enum.Font.Gotham
    btn.Parent = parent

    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 6)
    btnCorner.Parent = btn

    local isActive = false
    
    btn.MouseButton1Click:Connect(function()
        isActive = not isActive
        btn.Text = name .. (isActive and " [ON]" or " [OFF]")
        btn.BackgroundColor3 = isActive and Color3.fromRGB(60, 120, 100) or Color3.fromRGB(45, 45, 65)
        callback(isActive)
    end)

    return btn
end

-- KEYBIND
-- ============================================
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.RightShift then
        isUIVisible = not isUIVisible
        mainFrame.Visible = isUIVisible
        isCollecting = false
        isHovering = false
        print("UI toggled: " .. (isUIVisible and "ON" or "OFF"))
    end
end)

-- ============================================
-- MAIN LOOP
-- ============================================
RunService.Heartbeat:Connect(function()
    if not isUIVisible or not player.Character then return end

    if tick() - lastItemRefresh > 2 then
        itemCache = {}
        if ECS and ECS.Query then
            pcall(function()
                local items = ECS:Query("Item") or {}
                for _, item in pairs(items) do
                    if item and item.Parent then
                        table.insert(itemCache, item)
                    end
                end
            end)
        end
        lastItemRefresh = tick()
    end

    pcall(function()
        itemCountLabel.Text = "Items: " .. #itemCache .. " | Shards: " .. #shardCache
        statusLabel.Text = "Status: " .. 
            (isCollecting and "Collecting" or isHovering and "Hovering" or "Ready") ..
            "\nItems: " .. #itemCache ..
            "\nEntities: " .. #entityCache
    end)

    if isCollecting and player.Character then
        for _, item in pairs(itemCache) do
            if item and item.Parent then
                pcall(function()
                    item:MoveTo(player.Character.HumanoidRootPart.Position)
                end)
            end
        end
    end

    if isHovering and itemCache[1] and player.Character then
        pcall(function()
            local target = itemCache[1]
            if target and target.Parent then
                player.Character.HumanoidRootPart.CFrame = target.CFrame + Vector3.new(0, hoverOffset, 3)
                
                if cameraLocked then
                    camera.CFrame = CFrame.new(camera.Position, target.Position)
                end
            end
        end)
    end
end)
