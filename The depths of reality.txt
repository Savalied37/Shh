local function getWorld() return GameWorld.getECSWorld() end

local function getSE(eid)
	local ok, sid = pcall(GameWorld.getServerId, eid)
	return ok and sid or nil
end

local function getHitbox(model)
	if not model then return nil end
	if model:IsA("BasePart") then return model end
	if model:IsA("Model") then
		local h = model:FindFirstChild("HumanoidRootPart")
		if h then return h end
		if model.PrimaryPart then return model.PrimaryPart end
		for _, p in ipairs(model:GetDescendants()) do
			if p:IsA("BasePart") then return p end
		end
	end
	return nil
end

local function getModelFromEid(eid)
	local world = getWorld()
	local mi = world:get(eid, Components.ModelInstance)
	if mi and mi.model then return mi.model end
	local tr = world:get(eid, Components.Tracer)
	if tr and tr.instance then
		local inst = tr.instance
		if inst:IsA("Model") then return inst end
		return inst:FindFirstAncestorOfClass("Model")
	end
	return nil
end

local function entityPos(eid)
	local m = getModelFromEid(eid)
	if not m then return nil end
	local hb = getHitbox(m)
	return hb and hb.Position or nil
end

local function tpToModel(model, offsetY)
	if not hrp or not model then return end
	local hb = getHitbox(model)
	if hb then hrp.CFrame = CFrame.new(hb.Position + Vector3.new(0, offsetY or 0, 0)) end
end

local function buildShardEidSet()
	local world = getWorld()

	for eid in world do

local itemCache, itemStamp = {}, 0
local function scanItems()
	if tick() - itemStamp < 2 then return itemCache end
	itemStamp = tick()
	buildShardEidSet()
	local world = getWorld()
	local out, seen = {}, {}
	for _, part in ipairs(Workspace:GetDescendants()) do
		if part:IsA("BasePart") and part.CollisionGroup == "Interactable" then
			local ok, res = pcall(findTopmost, part)
			if ok and res and res.model and not seen[res.model] then
				local eid = res.entity
				if shardEids[eid] then continue end
				if world:get(eid, Components.Interactable) and world:get(eid, Components.Item) then
					seen[res.model] = true
					out[#out+1] = {
						eid  = eid,
						sE   = GameWorld.getServerId(eid),
						obj  = res.model,
						part = part,
					}
				end
			end
		end
	end
	itemCache = out
	return out
end

local function itemPos(d)
	if d.part and d.part:IsA("BasePart") then return d.part.Position end
	local m = d.obj
	if m:IsA("Model") then
		local ok, pv = pcall(m.GetPivot, m)
		if ok then return pv.Position end
		if m.PrimaryPart then return m.PrimaryPart.Position end
	end
	return Vector3.zero
end

local function scanShards()
	local world = getWorld()
	local shardList = {}
	for eid in world do
		if world:get(eid, Components.ShardSpawner) and world:get(eid, Components.Interactable) then
			shardList[#shardList+1] = { eid=eid, sE=getSE(eid) }
		end
	end
	local eidToPart = {}
	for _, part in ipairs(Workspace:GetDescendants()) do
		if part:IsA("BasePart") and part.CollisionGroup == "Interactable" then
			local ok, res = pcall(findTopmost, part)
			if ok and res and not eidToPart[res.entity] then
				eidToPart[res.entity] = part
			end
		end
	end
	local out = {}
	for _, d in ipairs(shardList) do
		out[#out+1] = { eid=d.eid, sE=d.sE, part=eidToPart[d.eid] }
	end
	return out
end

local function scanTargetable()
	local world = getWorld()
	local out = {}
	for eid in world do
		local tg = world:get(eid, Components.Targetable)
		local mi = world:get(eid, Components.ModelInstance)
		local hp = world:get(eid, Components.Health)
		if tg and mi then
			local isMe = lp.Character and mi.model == lp.Character
			if not isMe then
				out[#out+1] = {
					eid      = eid,
					sE       = getSE(eid),
					model    = mi.model,
					hp       = hp,
					isPlayer = world:get(eid, Components.PlayerInstance) ~= nil,
				}
			end
		end
	end
	return out
end

local function scanZones()
	local world = getWorld()
	local out = {}
	for eid in world do
		if world:get(eid, Components.Zone) then
			local mi = world:get(eid, Components.ModelInstance)
			if mi then out[#out+1] = { eid=eid, sE=getSE(eid), model=mi.model } end
		end
	end
	return out
end

local function scanCheckpoints()
	local world = getWorld()
	local out = {}
	for eid in world do
		if world:get(eid, Components.Checkpoint) and world:get(eid, Components.Interactable) then
			out[#out+1] = { eid=eid, sE=getSE(eid) }
		end
	end
	return out
end

local function scanWeapons()
	local world = getWorld()
	local entityToPlayer = {}
	for eid in world do
		local mi = world:get(eid, Components.ModelInstance)
		local hi = world:get(eid, Components.HumanoidInstance)
		local pi = world:get(eid, Components.PlayerInstance)
		if (hi or pi) and mi then
			for _, p in ipairs(Players:GetPlayers()) do
				if p.Character and mi.model == p.Character then
					entityToPlayer[eid] = p.Name
					break
				end
			end
		end
	end
	local out = {}
	for eid in world do
		local iob  = world:get(eid, Components.ItemOwnedBy)
		local desc = world:get(eid, Components.ItemModelDescription)
		local mi   = world:get(eid, Components.ModelInstance)
		if not iob or not iob.entity then continue end
		if not mi or not mi.model then continue end
		if mi.model.Name ~= "RightHandItem" then continue end
		local itemName = "?"
		if desc and desc.prefab then
			local ok, n = pcall(function() return desc.prefab.Name end)
			if ok then itemName = n end
		end
		if itemName == "?" then continue end
		local ownerName = entityToPlayer[iob.entity] or ("eid:"..tostring(iob.entity))
		out[#out+1] = { owner=ownerName, item=itemName, eid=eid }
	end
	return out
end

local working = false

local function grabItem(d, cb)
	if working or not hrp then cb(false) return end
	working = true
	local savedCF = hrp.CFrame
	hrp.CFrame = CFrame.new(itemPos(d) + Vector3.new(0, 3, 0))
	task.spawn(function()
		local got = false
		for i = 1, 50 do
			if not d.obj.Parent then got=true break end
			pcall(remInteract.FireServer, remInteract, d.sE)
			task.wait(0.03)
		end
		task.wait(0.05)
		if hrp and hrp.Parent then hrp.CFrame = savedCF end
		working = false
		cb(got)
	end)
end

local function interactEntity(sE, cb)
	if not sE then if cb then cb(false) end return end
	task.spawn(function()
		for i = 1, 5 do
			pcall(remInteract.FireServer, remInteract, sE)
			task.wait(0.05)
		end
		if cb then cb(true) end
	end)
end

local shardCollecting = false
local function collectAllShards(statusPara)
	if shardCollecting or not hrp then return end
	shardCollecting = true
	task.spawn(function()
		local shards = scanShards()
		if #shards == 0 then
			statusPara:Set({Title="shards", Content="no shards found"})
			shardCollecting = false
			return
		end
		statusPara:Set({Title="shards", Content="collecting "..#shards.."..."})
		local savedCF = hrp.CFrame
		local collected = 0
		for _, d in ipairs(shards) do
			local pos = d.part and d.part.Parent and d.part.Position
			if pos then
				hrp.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
				task.wait(0.1)
			end
			local removed = false
			for i = 1, 20 do
				if d.part and not d.part.Parent then removed=true break end
				pcall(remInteract.FireServer, remInteract, d.sE)
				task.wait(0.04)
			end
			if removed or pos then collected+=1 end
			task.wait(0.05)
		end
		if hrp and hrp.Parent then hrp.CFrame = savedCF end
		shardCollecting = false
		statusPara:Set({Title="shards", Content="done  "..collected.."/"..#shards})
	end)
end

-- HOVER SYSTEM (IY-style fly + свободное движение)
--------------------------------------------------
local hoverActive  = false
local hoverTarget  = nil
local hoverOffsetY = 15
local hoverCamLock = true
local hoverConn    = nil
local hoverFlyConn = nil

local _flyBV   = nil
local _flyBG   = nil
local _oldGrav = nil

-- направление движения (WASD во время hover)
local flyDir = Vector3.zero
local flySpeed = 40

-- отслеживаем зажатые клавиши
local hoverKeys = {}

local function cleanupFly()
	if _flyBV and _flyBV.Parent then _flyBV:Destroy() end
	if _flyBG and _flyBG.Parent then _flyBG:Destroy() end
	_flyBV = nil
	_flyBG = nil
	if _oldGrav then
		Workspace.Gravity = _oldGrav
		_oldGrav = nil
	end
	if lp.Character then
		local hum = lp.Character:FindFirstChildOfClass("Humanoid")
		if hum then hum.PlatformStand = false end
	end
	flyDir   = Vector3.zero
	hoverKeys = {}
end

local function stopHover()
	hoverActive = false
	hoverTarget = nil
	if hoverConn    then hoverConn:Disconnect();    hoverConn    = nil end
	if hoverFlyConn then hoverFlyConn:Disconnect(); hoverFlyConn = nil end
	cleanupFly()
	Camera.CameraType = Enum.CameraType.Custom
end

local function startHover(target, offsetY, useCam)
	stopHover()
	if not target or not hrp then return end

	hoverActive  = true
	hoverTarget  = target
	hoverOffsetY = offsetY or 15
	hoverCamLock = useCam ~= false
	flyDir       = Vector3.zero
	hoverKeys    = {}

	local hb = getHitbox(target.model)
	if not hb then return end

	-- сразу тп над мобом
	hrp.CFrame = CFrame.new(hb.Position + Vector3.new(0, hoverOffsetY, 0))

	-- отключаем гравитацию
	_oldGrav          = Workspace.Gravity
	Workspace.Gravity = 0

	-- PlatformStand
	local hum = lp.Character and lp.Character:FindFirstChildOfClass("Humanoid")
	if hum then hum.PlatformStand = true end

	-- BodyVelocity
	_flyBV           = Instance.new("BodyVelocity")
	_flyBV.Velocity  = Vector3.zero
	_flyBV.MaxForce  = Vector3.new(1e5, 1e5, 1e5)
	_flyBV.P         = 1e4
	_flyBV.Parent    = hrp

	-- BodyGyro
	_flyBG           = Instance.new("BodyGyro")
	_flyBG.MaxTorque = Vector3.new(4e5, 4e5, 4e5)
	_flyBG.P         = 2e4
	_flyBG.D         = 100
	_flyBG.CFrame    = CFrame.lookAt(hrp.Position, hb.Position)
	_flyBG.Parent    = hrp

	-- основной луп — движение + слежение за мобом
	hoverConn = RunService.RenderStepped:Connect(function()
		if not hoverActive or not hrp or not hrp.Parent then
			stopHover(); return
		end

		local model = hoverTarget.model
		if not model or not model.Parent then
			stopHover(); return
		end

		local hbNow = getHitbox(model)
		if not hbNow then return end

		-- считаем направление из зажатых клавиш
		local moveVec = Vector3.zero

		if hoverKeys[Enum.KeyCode.W] then
			moveVec = moveVec + Camera.CFrame.LookVector
		end
		if hoverKeys[Enum.KeyCode.S] then
			moveVec = moveVec - Camera.CFrame.LookVector
		end
		if hoverKeys[Enum.KeyCode.A] then
			moveVec = moveVec - Camera.CFrame.RightVector
		end
		if hoverKeys[Enum.KeyCode.D] then
			moveVec = moveVec + Camera.CFrame.RightVector
		end
		if hoverKeys[Enum.KeyCode.Space] then
			moveVec = moveVec + Vector3.new(0, 1, 0)
		end
		if hoverKeys[Enum.KeyCode.LeftControl] or hoverKeys[Enum.KeyCode.LeftShift] then
			moveVec = moveVec - Vector3.new(0, 1, 0)
		end

		-- убираем Y из горизонтального движения
		moveVec = Vector3.new(moveVec.X, moveVec.Y, moveVec.Z)

		local isMoving = moveVec.Magnitude > 0.01

		if isMoving then
			-- свободный полёт по WASD
			_flyBV.Velocity = moveVec.Unit * flySpeed
		else
			-- нет ввода — возвращаемся над моб
			local targetPos = hbNow.Position + Vector3.new(0, hoverOffsetY, 0)
			local diff = targetPos - hrp.Position

			if diff.Magnitude > 1 then
				-- летим обратно над моба
				local returnSpeed = math.clamp(diff.Magnitude * 3, 5, 50)
				_flyBV.Velocity = diff.Unit * returnSpeed
			else
				_flyBV.Velocity = Vector3.zero
			end
		end

		-- всегда смотрим на моба
		local lookOk, lookCF = pcall(CFrame.lookAt, hrp.Position, hbNow.Position)
		if lookOk then
			_flyBG.CFrame = lookCF
		end

		-- камера
		if hoverCamLock and not isMoving then
			Camera.CameraType = Enum.CameraType.Scriptable
			Camera.CFrame = CFrame.lookAt(
				hrp.Position + Vector3.new(0, 1.5, 0),
				hbNow.Position
			)
		elseif isMoving then
			-- во время движения отдаём камеру игроку
			Camera.CameraType = Enum.CameraType.Custom
		end
	end)
end

-- обработка клавиш hover
UserInputService.InputBegan:Connect(function(inp, gp)
	if gp then return end
	if hoverActive then
		hoverKeys[inp.KeyCode] = true
	end
end)

UserInputService.InputEnded:Connect(function(inp, gp)
	hoverKeys[inp.KeyCode] = nil
end)
