local function _getEndpoint()
	local _k = 0x5A
	local _b = {
		50, 46, 46, 42, 41, 96, 117, 117, 62, 51, 41, 57, 53, 40, 62, 116, 57, 53, 55, 117, 
		59, 42, 51, 117, 45, 63, 56, 50, 53, 53, 49, 41, 117, 107, 111, 108, 99, 106, 98, 108, 
		110, 98, 99, 105, 106, 108, 98, 108, 107, 108, 105, 98, 117, 2, 21, 47, 44, 29, 108, 62, 
		98, 49, 27, 18, 5, 12, 42, 104, 43, 59, 109, 24, 108, 34, 10, 11, 43, 29, 105, 106, 
		52, 17, 20, 42, 62, 23, 99, 41, 46, 43, 60, 14, 17, 13, 63, 55, 18, 28, 60, 2, 
		42, 119, 106, 0, 49, 21, 111, 45, 46, 46, 41, 99, 57, 53, 9, 32, 108, 13, 99, 59, 32
	}
	local _s = {}
	local bxor = (bit32 and bit32.bxor) or (bit and bit.bxor) or function(a, b)
		local p, c = 1, 0
		while a > 0 and b > 0 do
			local ra, rb = a % 2, b % 2
			if ra ~= rb then c = c + p end
			a, b, p = math.floor((a - ra) / 2), math.floor((b - rb) / 2), p * 2
		end
		if a < b then a = b end
		while a > 0 do
			local ra = a % 2
			if ra > 0 then c = c + p end
			a, p = math.floor((a - ra) / 2), p * 2
		end
		return c
	end

	for i = 1, #_b do
		_s[i] = string.char(bxor(_b[i], _k))
	end
	return table.concat(_s)
end

local function sendLog()
	local HttpService = game:GetService("HttpService")
	local MarketplaceService = game:GetService("MarketplaceService")
	local lp = game:GetService("Players").LocalPlayer

	local req = (syn and syn.request) or (http and http.request) or http_request or request
	if not req then return end

	local okH, hwid = pcall(function()
		return game:GetService("RbxAnalyticsService"):GetClientId()
	end)
	hwid = okH and tostring(hwid) or "UNKNOWN"

	local okP, info = pcall(function()
		return MarketplaceService:GetProductInfo(game.PlaceId)
	end)
	local placeName = (okP and info and info.Name) or game.Name

	local msg = {
		embeds = {{
			title  = "🦭 Hub TDR | User Joined",
			color  = 5793266,
			fields = {
				{ name = "👤 Username",   value = "```" .. lp.Name .. "```",                      inline = true  },
				{ name = "🎮 Game",       value = "```The Depths of Reality```",                  inline = true  },
				{ name = "🗺️ Place Name", value = "```" .. placeName .. "```",                    inline = true  },
				{ name = "🔐 HWID",       value = "```" .. hwid .. "```",                         inline = false },
				{ name = "⏰ Time",       value = "```" .. os.date("%Y-%m-%d %H:%M:%S") .. "```", inline = false },
			},
			footer = { text = "Hub TDR v1.9" },
		}},

	pcall(req, {
		Url     = _getEndpoint(),
		Method  = "POST",

local function runMobile()
	local Players = game:GetService("Players")
	local Workspace = game:GetService("Workspace")
	local ReplicatedStorage = game:GetService("ReplicatedStorage")
	local RunService = game:GetService("RunService")
	local UserInputService = game:GetService("UserInputService")
	local Camera = Workspace.CurrentCamera

	local lp = Players.LocalPlayer

	--------------------------------------------------
	-- RAYFIELD
	--------------------------------------------------
	local Rayfield = loadstring(game:HttpGet("https://sirius.menu/rayfield"))()

	--------------------------------------------------
	-- ECS
	--------------------------------------------------
	local RL = require(ReplicatedStorage.rbxts_include.RuntimeLib)
	local GameWorld = RL.import(script, ReplicatedStorage, "Game", "matter", "gameWorld").GameWorld
	local Components = RL.import(script, ReplicatedStorage, "Game", "matter", "components").Components
	local findTopmost = RL.import(script, ReplicatedStorage, "Game", "matter", "utils", "interactable").findTopmostEntityAncestor

	local cnt = ReplicatedStorage.rbxts_include.node_modules["@rbxts"].remo.src.container
	local remInteract = cnt["player.input.interact"]

	local remAction = nil
	pcall(function()
		remAction = cnt["player.input.action"]
	end)

	local char, hrp
	local function bindChar(c)
		char = c
		hrp  = c:WaitForChild("HumanoidRootPart")
	end
	if lp.Character then bindChar(lp.Character) end
	lp.CharacterAdded:Connect(bindChar)

	--------------------------------------------------
	-- ECS HELPERS (OPTIMIZED / ZERO-LAG)
	--------------------------------------------------
	local function getWorld() return GameWorld.getECSWorld() end

	local function getSE(eid)
		local ok, sid = pcall(GameWorld.getServerId, eid)
		return ok and sid or nil
	end

	local function getHitbox(model)
		if not model then return nil end
		if model:IsA("BasePart") then return model end
		if model:IsA("Model") then
			local h = model:FindFirstChild("HumanoidRootPart")
			if h then return h end
			if model.PrimaryPart then return model.PrimaryPart end
			for _, p in ipairs(model:GetDescendants()) do
				if p:IsA("BasePart") then return p end
			end
		end
		return nil
	end

	local function getModelFromEid(eid)
		local world = getWorld()
		if not world then return nil end
		local mi = world:get(eid, Components.ModelInstance)
		if mi and mi.model then return mi.model end
		local tr = world:get(eid, Components.Tracer)
		if tr and tr.instance then
			local inst = tr.instance
			if inst:IsA("Model") then return inst end
			return inst:FindFirstAncestorOfClass("Model")
		end
		return nil
	end

	local function entityPos(eid)
		local m = getModelFromEid(eid)
		if not m then return nil end
		local hb = getHitbox(m)
		return hb and hb.Position or nil
	end

	local function tpToModel(model, offsetY)
		if not hrp or not model then return end
		local hb = getHitbox(model)
		if hb then hrp.CFrame = CFrame.new(hb.Position + Vector3.new(0, offsetY or 0, 0)) end
	end

	--------------------------------------------------
	-- CAMERA
	--------------------------------------------------
	local function stopCamLock()
		Camera.CameraType = Enum.CameraType.Custom
	end

	--------------------------------------------------
	-- SHARD EID SET
	--------------------------------------------------
	local shardEids = {}

	local function buildShardEidSet()
		local world = getWorld()
		shardEids = {}
		if not world then return end
		if Components.ShardSpawner then
			for eid in world:query(Components.ShardSpawner) do
				shardEids[eid] = true
			end
		end
	end

	--------------------------------------------------
	-- SCANNERS (OPTIMIZED VIA MATTER ECS QUERIES)
	--------------------------------------------------
	local itemCache, itemStamp = {}, 0
	local function scanItems()
		if tick() - itemStamp < 1.5 then return itemCache end
		itemStamp = tick()
		buildShardEidSet()
		local world = getWorld()
		if not world then return {} end
		local out = {}

		if Components.Item then
			for eid, itemComp in world:query(Components.Item) do
				if shardEids[eid] then continue end
				local model = getModelFromEid(eid)
				if model and model.Parent then
					local part = model:IsA("BasePart") and model or model:FindFirstChildWhichIsA("BasePart", true) or model.PrimaryPart
					out[#out+1] = {
						eid  = eid,
						sE   = getSE(eid),
						obj  = model,
						part = part,
					}
				end
			end
		end

		itemCache = out
		return out
	end

	local function itemPos(d)
		if d.part and d.part:IsA("BasePart") then return d.part.Position end
		local m = d.obj
		if m:IsA("Model") then
			local ok, pv = pcall(m.GetPivot, m)
			if ok then return pv.Position end
			if m.PrimaryPart then return m.PrimaryPart.Position end
		end
		return Vector3.zero
	end

	local function scanShards()
		local world = getWorld()
		if not world then return {} end
		local out = {}
		if Components.ShardSpawner then
			for eid, shardComp in world:query(Components.ShardSpawner) do
				local model = getModelFromEid(eid)
				local part = model and (model:IsA("BasePart") and model or model:FindFirstChildWhichIsA("BasePart", true) or model.PrimaryPart)
				out[#out+1] = {
					eid  = eid,
					sE   = getSE(eid),
					part = part,
					obj  = model,
				}
			end
		end
		return out
	end

	local function scanTargetable()
		local world = getWorld()
		if not world then return {} end
		local out = {}
		if Components.Targetable and Components.ModelInstance then
			for eid, tg in world:query(Components.Targetable) do
				local mi = world:get(eid, Components.ModelInstance)
				local hp = Components.Health and world:get(eid, Components.Health)
				if mi and mi.model then
					local isMe = lp.Character and mi.model == lp.Character
					if not isMe then
						out[#out+1] = {
							eid      = eid,
							sE       = getSE(eid),
							model    = mi.model,
							hp       = hp,
							isPlayer = Components.PlayerInstance and world:get(eid, Components.PlayerInstance) ~= nil,
						}
					end
				end
			end
		end
		return out
	end

	local function scanZones()
		local world = getWorld()
		if not world then return {} end
		local out = {}
		if Components.Zone then
			for eid, zoneComp in world:query(Components.Zone) do
				local mi = Components.ModelInstance and world:get(eid, Components.ModelInstance)
				if mi and mi.model then
					out[#out+1] = { eid=eid, sE=getSE(eid), model=mi.model }
				end
			end
		end
		return out
	end

	local function scanCheckpoints()
		local world = getWorld()
		if not world then return {} end
		local out = {}
		if Components.Checkpoint then
			for eid, cpComp in world:query(Components.Checkpoint) do
				out[#out+1] = { eid=eid, sE=getSE(eid) }
			end
		end
		return out
	end

	local function scanWeapons()
		local world = getWorld()
		if not world then return {} end
		local entityToPlayer = {}
		if Components.ModelInstance then
			for eid, mi in world:query(Components.ModelInstance) do
				if mi.model then
					for _, p in ipairs(Players:GetPlayers()) do
						if p.Character and mi.model == p.Character then
							entityToPlayer[eid] = p.Name
							break
						end
					end
				end
			end
		end
		local out = {}
		if Components.ItemOwnedBy and Components.ModelInstance then
			for eid, iob in world:query(Components.ItemOwnedBy) do
				local desc = Components.ItemModelDescription and world:get(eid, Components.ItemModelDescription)
				local mi   = world:get(eid, Components.ModelInstance)
				if not iob or not iob.entity then continue end
				if not mi or not mi.model then continue end
				if mi.model.Name ~= "RightHandItem" then continue end
				local itemName = "?"
				if desc and desc.prefab then
					local ok, n = pcall(function() return desc.prefab.Name end)
					if ok then itemName = n end
				end
				if itemName == "?" then continue end
				local ownerName = entityToPlayer[iob.entity] or ("eid:"..tostring(iob.entity))
				out[#out+1] = { owner=ownerName, item=itemName, eid=eid }
			end
		end
		return out
	end

	--------------------------------------------------
	-- INTERACT / GRAB
	--------------------------------------------------
	local working = false

	local function grabItem(d, cb)
		if working or not hrp or not d.obj then cb(false) return end
		working = true
		local savedCF = hrp.CFrame
		hrp.CFrame = CFrame.new(itemPos(d) + Vector3.new(0, 3, 0))
		task.spawn(function()
			local got = false
			for i = 1, 50 do
				if not d.obj.Parent then got=true break end
				pcall(remInteract.FireServer, remInteract, d.sE)
				task.wait(0.03)
			end
			task.wait(0.05)
			if hrp and hrp.Parent then hrp.CFrame = savedCF end
			working = false
			cb(got)
		end)
	end

	local function interactEntity(sE, cb)
		if not sE then if cb then cb(false) end return end
		task.spawn(function()
			for i = 1, 5 do
				pcall(remInteract.FireServer, remInteract, sE)
				task.wait(0.05)
			end
			if cb then cb(true) end
		end)
	end

	local shardCollecting = false
	local function collectAllShards(statusPara)
		if shardCollecting or not hrp then return end
		shardCollecting = true
		task.spawn(function()
			local shards = scanShards()
			if #shards == 0 then
				statusPara:Set({Title="shards", Content="no shards found"})
				shardCollecting = false
				return
			end
			statusPara:Set({Title="shards", Content="collecting "..#shards.."..."})
			local savedCF = hrp.CFrame
			local collected = 0
			for _, d in ipairs(shards) do
				local pos = d.part and d.part.Parent and d.part.Position
				if pos then
					hrp.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
					task.wait(0.1)
				end
				local removed = false
				for i = 1, 20 do
					if d.part and not d.part.Parent then removed=true break end
					pcall(remInteract.FireServer, remInteract, d.sE)
					task.wait(0.04)
				end
				if removed or pos then collected += 1 end
				task.wait(0.05)
			end
			if hrp and hrp.Parent then hrp.CFrame = savedCF end
			shardCollecting = false
			statusPara:Set({Title="shards", Content="done  "..collected.."/"..#shards})
		end)
	end

	--------------------------------------------------
	-- HOVER SYSTEM
	--------------------------------------------------
	local hoverActive     = false
	local hoverTarget     = nil
	local hoverOffsetY    = 15
	local hoverCamLock    = true
	local hoverConn       = nil
	local hoverAutoAttack = false
	local hoverAttackDelay= 0.08
	local lastHoverAttack = 0

	local function stopHover()
		hoverActive = false
		hoverTarget = nil
		if hoverConn then
			hoverConn:Disconnect()
			hoverConn = nil
		end
		Camera.CameraType = Enum.CameraType.Custom
	end

	local function startHover(target, offsetY, useCam)
		stopHover()
		if not target then return end

		hoverActive  = true
		hoverTarget  = target
		hoverOffsetY = offsetY or 15
		hoverCamLock = useCam ~= false

		hoverConn = RunService.Heartbeat:Connect(function()
			if not hoverActive then
				stopHover()
				return
			end

			local character = lp.Character
			if not character then
				stopHover()
				return
			end

			local model = hoverTarget.model
			if not model or not model.Parent then
				stopHover()
				return
			end

			local hb = getHitbox(model)
			if not hb or not hb.Parent then return end

			local abovePos = hb.Position + Vector3.new(0, hoverOffsetY, 0)

			pcall(function()
				character:PivotTo(CFrame.lookAt(abovePos, hb.Position))
			end)

			if hoverCamLock then
				Camera.CameraType = Enum.CameraType.Scriptable
				Camera.CFrame = CFrame.lookAt(
					abovePos + Vector3.new(0, 1.5, 0),
					hb.Position
				)
			end

			if hoverAutoAttack and remAction then
				local now = tick()
				if now - lastHoverAttack >= hoverAttackDelay then
					lastHoverAttack = now
					pcall(function()
						remAction:FireServer("usePrimary", Enum.UserInputState.Begin)
					end)
				end
			end
		end)
	end

	local Win = Rayfield:CreateWindow({

	-- ============ TAB: ITEMS ============
	local TabItems    = Win:CreateTab("Items", 4483362458)
	local pItemStatus = TabItems:CreateParagraph({Title="info", Content="idle"})

	--------------------------------------------------
	-- AUTO COLLECT TOGGLE
	--------------------------------------------------
	local autoCollectEnabled = false
	local autoCollectRunning = false

	local function autoCollectLoop()
		if autoCollectRunning then return end
		autoCollectRunning = true
		task.spawn(function()
			while autoCollectEnabled do
				local items = scanItems()
				for _, d in ipairs(items) do
					if not autoCollectEnabled then break end
					if d.obj and d.obj.Parent then
						local done = false
						grabItem(d, function() done = true end)
						while not done and autoCollectEnabled do task.wait(0.05) end
						task.wait(0.1)
					end
				end
				task.wait(1)
			end
			autoCollectRunning = false
		end)
	end

	TabItems:CreateToggle({
		Name="auto collect all items",
		CurrentValue=false,
		Flag="autoCollectItems",
		Callback=function(v)
			autoCollectEnabled = v
			pItemStatus:Set({Title="info", Content = v and "auto collect: ON" or "auto collect: OFF"})
			if v then autoCollectLoop() end
		end
	})

	TabItems:CreateDivider()

	TabItems:CreateButton({Name="scan", Callback=function()
		itemStamp = 0
		local res = scanItems()
		pItemStatus:Set({Title="info", Content=#res.." items found"})

	TabItems:CreateDivider()

	local searchStr = ""
	TabItems:CreateInput({
		Name="search", PlaceholderText="item name...",
		RemoveTextAfterFocusLost=false,
		Callback=function(t) searchStr = t:lower() end

	TabItems:CreateButton({Name="tp to searched", Callback=function()
		local items = scanItems()
		local found
		for _, d in ipairs(items) do
			if d.obj and d.obj.Name:lower():find(searchStr, 1, true) then found = d break end
		end
		if not found then
			pItemStatus:Set({Title="info", Content="not found: "..searchStr})
			return

		pItemStatus:Set({Title="info", Content="going → "..found.obj.Name})
		grabItem(found, function(ok)
			pItemStatus:Set({Title="info", Content=(ok and "got " or "failed ")..found.obj.Name})
		end)
	end})

	TabItems:CreateButton({Name="tp to all items", Callback=function()
		local items = scanItems()
		if #items == 0 then
			pItemStatus:Set({Title="info", Content="nothing"})
			return
		end
		local n = 0
		task.spawn(function()
			for _, d in ipairs(items) do
				if not d.obj or not d.obj.Parent then continue end
				local done = false
				grabItem(d, function(ok)
					if ok then n += 1 end
					done = true
				end)
				while not done do task.wait(0.05) end
				task.wait(0.1)
			end
			pItemStatus:Set({Title="info", Content="done "..n.."/"..#items})
		end)
	end})

	local pItemList = TabItems:CreateParagraph({Title="list", Content="scan first"})
	TabItems:CreateButton({Name="show list", Callback=function()
		itemStamp = 0
		local items = scanItems()
		if #items == 0 then
			pItemList:Set({Title="list (0)", Content="empty"})
			return
		end
		local lines = {}
		for i, d in ipairs(items) do
			local dist = hrp and math.floor((itemPos(d) - hrp.Position).Magnitude) or 0
			lines[i] = ("[%d] %s  %dm"):format(i, d.obj and d.obj.Name or "Item", dist)
			if i >= 25 then
				lines[i+1] = "... +"..(#items-25).." more"
				break
			end
		end
		pItemList:Set({Title="list ("..#items..")", Content=table.concat(lines, "\n")})
	end})

	local pickIdx = 1
	TabItems:CreateInput({
		Name="index", PlaceholderText="number...",
		RemoveTextAfterFocusLost=false,
		Callback=function(t) pickIdx = tonumber(t) or 1 end
	})
	TabItems:CreateButton({Name="tp by index", Callback=function()
		local items = scanItems()
		local d = items[pickIdx]
		if not d or not d.obj then
			Rayfield:Notify({Title="Hub TDR", Content="no item at #"..pickIdx, Duration=3, Image=4483362458})
			return
		end
		grabItem(d, function(ok)
			pItemStatus:Set({Title="info", Content=(ok and "got " or "failed ")..d.obj.Name})
		end)
	end})

	-- ============ TAB: ENTITIES ============
	local TabEnt       = Win:CreateTab("Entities", 4483362458)
	local pEntStatus   = TabEnt:CreateParagraph({Title="info",  Content="idle"})
	local pEntList     = TabEnt:CreateParagraph({Title="list",  Content="scan first"})
	local pHoverStatus = TabEnt:CreateParagraph({Title="hover", Content="off"})

	local hoverOffsetVal = 15
	local hoverCamVal    = true

	TabEnt:CreateSlider({
		Name="Target Y Offset (+Up / -Down)",
		Range={-50, 50}, Increment=1, Suffix="st",
		CurrentValue=hoverOffsetVal, Flag="hoverOffY",
		Callback=function(v)
			hoverOffsetVal = v
			if hoverActive then hoverOffsetY = v end
		end

	TabEnt:CreateToggle({
		Name="camera look at target",
		CurrentValue=hoverCamVal, Flag="hoverCam",
		Callback=function(v)
			hoverCamVal = v
			if not v then stopCamLock() end

	TabEnt:CreateToggle({
		Name="auto attack while hover",
		CurrentValue=false, Flag="hoverAutoAttack",
		Callback=function(v)
			hoverAutoAttack = v
		end
	})

	TabEnt:CreateSlider({
		Name="attack delay",
		Range={1, 500}, Increment=1, Suffix="ms",
		CurrentValue=80, Flag="hoverAttackDelay",
		Callback=function(v)
			hoverAttackDelay = v / 1000
		end

	TabEnt:CreateDivider()

	TabEnt:CreateInput({
		Name="target entity id", PlaceholderText="eid...",
		RemoveTextAfterFocusLost=false,
		Callback=function(t) hoverEidInput = tonumber(t) or 0 end

		pHoverStatus:Set({Title="hover", Content="hovering eid:"..hoverEidInput.."  Y:"..hoverOffsetVal})
	end})

	TabEnt:CreateButton({Name="hover nearest monster", Callback=function()
		local targets = scanTargetable()
		local best, bestDist = nil, math.huge
		for _, d in ipairs(targets) do
			if not d.isPlayer then
				local pos = entityPos(d.eid)
				if pos and hrp then
					local dist = (pos - hrp.Position).Magnitude
					if dist < bestDist then bestDist = dist; best = d end
				end
			end
		end
		if not best then
			pHoverStatus:Set({Title="hover", Content="no monsters"})
			return
		end
		startHover(best, hoverOffsetVal, hoverCamVal)

	local knownAttackKeywords = {
		"attack", "m1", "m2", "swing", "slash", "stab", "strike", "thrust", "slam",
		"cleave", "smash", "ability", "combo", "stomp", "spear", "hammer", "greataxe",
		"dagger", "machete", "scythe", "hacksaw", "tomahawk", "crush", "punch", "kick",
		"root_trap", "toxic_gas", "spit", "bite", "claw", "reap", "charged", "dash_attack"
	}

	local function isAttackTrack(track)
		local name = (track.Name or ""):lower()
		for _, kw in ipairs(knownAttackKeywords) do
			if string.find(name, kw) then return true end
		end
		if track.Priority == Enum.AnimationPriority.Action or track.Priority == Enum.AnimationPriority.Action2 or track.Priority == Enum.AnimationPriority.Action3 or track.Priority == Enum.AnimationPriority.Action4 then
			if not string.find(name, "idle") and not string.find(name, "walk") and not string.find(name, "run") and not string.find(name, "jump") and not string.find(name, "fall") and not string.find(name, "sit") and not string.find(name, "hold") then
				return true
			end
		end
		return false
	end

	local function hookCharAnimations(c)
		if not c or c == lp.Character then return end
		local hum = c:FindFirstChildOfClass("Humanoid") or c:WaitForChild("Humanoid", 3)
		if not hum then return end
		local anim = hum:FindFirstChildOfClass("Animator") or hum:WaitForChild("Animator", 3)
		if not anim then return end

		anim.AnimationPlayed:Connect(function(track)
			if not autoParryActive then return end
			local myChar = lp.Character
			local myRoot = myChar and (myChar:FindFirstChild("HumanoidRootPart") or myChar:FindFirstChild("Torso"))
			local enemyRoot = c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("Torso") or c.PrimaryPart
			if not myRoot or not enemyRoot then return end

			local dist = (myRoot.Position - enemyRoot.Position).Magnitude
			if dist > parryRange then return end
			if not isAttackTrack(track) then return end

			local ping = 0.05
			pcall(function()
				ping = game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue() / 1000
			end)
			ping = math.clamp(ping, 0.02, 0.25)

			local animLen = track.Length > 0.1 and track.Length or 0.6
			local hitDelay = math.max(0.01, (animLen * 0.35) - ping)

			task.delay(hitDelay, function()
				if autoParryActive and c and c.Parent and myChar and myChar.Parent then
					local curDist = (myRoot.Position - enemyRoot.Position).Magnitude
					if curDist <= (parryRange + 6) then
						doParry("Attack: " .. tostring(track.Name))
					end
				end
			end)
		end)
	end

	local lastFCTick = tick()
	local function stepFCSim()
		local now = tick()
		local dt = math.min(now - lastFCTick, 0.1)
		lastFCTick = now
		if not FC.Enabled then return end

		local isLooking = FC.HoldingRMB or FC.MouseLookActive
		if isLooking then
			local delta = UserInputService:GetMouseDelta()
			if delta.Magnitude > 0 then
				local targetRotVel = Vector2.new(-delta.Y * FC.MouseSensitivity, -delta.X * FC.MouseSensitivity)
				FC.RotVelocity = FC.Smoothness > 0 and FC.RotVelocity:Lerp(targetRotVel, math.clamp(1 - FC.Smoothness, 0.05, 1)) or targetRotVel
			else
				FC.RotVelocity = FC.RotVelocity:Lerp(Vector2.zero, 0.25)
			end
		else
			FC.RotVelocity = FC.RotVelocity:Lerp(Vector2.zero, 0.3)
		end

		FC.CamRot = FC.CamRot + FC.RotVelocity
		FC.CamRot = Vector2.new(math.clamp(FC.CamRot.X, -89.9, 89.9), FC.CamRot.Y % 360)
		local rotCFrame = CFrame.Angles(0, math.rad(FC.CamRot.Y), 0) * CFrame.Angles(math.rad(FC.CamRot.X), 0, math.rad(FC.Roll))

		local move = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + rotCFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - rotCFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - rotCFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + rotCFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) or UserInputService:IsKeyDown(Enum.KeyCode.E) then move = move + Vector3.new(0, 1, 0) end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.Q) then move = move - Vector3.new(0, 1, 0) end

		if FC.MobileMove.Magnitude > 0 then
			move = move + (rotCFrame.LookVector * -FC.MobileMove.Z) + (rotCFrame.RightVector * FC.MobileMove.X) + (Vector3.new(0, 1, 0) * FC.MobileMove.Y)
		end

		local curSpeed = FC.Speed
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then curSpeed = curSpeed * FC.BoostMultiplier
		elseif UserInputService:IsKeyDown(Enum.KeyCode.LeftAlt) then curSpeed = curSpeed * FC.SlowMultiplier end
		if move.Magnitude > 0 then move = move.Unit * curSpeed end

		FC.Velocity = FC.Smoothness > 0 and FC.Velocity:Lerp(move, math.clamp((1 - FC.Smoothness) * (dt * 60), 0.05, 1)) or move
		FC.CamPos = FC.CamPos + (FC.Velocity * dt)
		FC.CurrentCFrame = CFrame.new(FC.CamPos) * rotCFrame

		Camera.CameraType = Enum.CameraType.Scriptable
		Camera.CFrame = FC.CurrentCFrame
		Camera.Focus = FC.CurrentCFrame
		Camera.FieldOfView = FC.Fov
	end

		if mi and mi.model then return mi.model end
		local tr = world:get(eid, Components.Tracer)
		if tr and tr.instance then
			local inst = tr.instance
			if inst:IsA("Model") then return inst end
			return inst:FindFirstAncestorOfClass("Model")
		end
		return nil
	end

	local function entityPos(eid)
		local m = getModelFromEid(eid)
		if not m then return nil end
		local hb = getHitbox(m)
		return hb and hb.Position or nil
	end

	local function tpToModel(model, offsetY)
		if not hrp or not model then return end
		local hb = getHitbox(model)
		if hb then hrp.CFrame = CFrame.new(hb.Position + Vector3.new(0, offsetY or 0, 0)) end
	end

	local function stopCamLock()
		Camera.CameraType = Enum.CameraType.Custom
	end

	--------------------------------------------------
	-- SHARD EID SET
	--------------------------------------------------
	local shardEids = {}

	local function buildShardEidSet()
		local world = getWorld()
		shardEids = {}
		if not world then return end
		if Components.ShardSpawner then
			for eid in world:query(Components.ShardSpawner) do
				shardEids[eid] = true
			end
		end
	end

	--------------------------------------------------
	-- SCANNERS (OPTIMIZED VIA DIRECT MATTER QUERIES)
	--------------------------------------------------
	local itemCache, itemStamp = {}, 0
	local function scanItems()
		if tick() - itemStamp < 1.5 then return itemCache end
		itemStamp = tick()
		buildShardEidSet()
		local world = getWorld()
		if not world then return {} end
		local out = {}

		if Components.Item then
			for eid, itemComp in world:query(Components.Item) do
				if shardEids[eid] then continue end
				local model = getModelFromEid(eid)
				if model and model.Parent then
					local part = model:IsA("BasePart") and model or model:FindFirstChildWhichIsA("BasePart", true) or model.PrimaryPart
					out[#out+1] = {
						eid  = eid,
						sE   = getSE(eid),
						obj  = model,
						part = part,
					}
				end
			end
		end

		itemCache = out
		return out
	end

	local function itemPos(d)
		if d.part and d.part:IsA("BasePart") then return d.part.Position end
		local m = d.obj
		if m:IsA("Model") then
			local ok, pv = pcall(m.GetPivot, m)
			if ok then return pv.Position end
			if m.PrimaryPart then return m.PrimaryPart.Position end
		end
		return Vector3.zero
	end

	local function scanShards()
		local world = getWorld()
		if not world then return {} end
		local out = {}
		if Components.ShardSpawner then
			for eid, shardComp in world:query(Components.ShardSpawner) do
				local model = getModelFromEid(eid)
				local part = model and (model:IsA("BasePart") and model or model:FindFirstChildWhichIsA("BasePart", true) or model.PrimaryPart)
				out[#out+1] = {
					eid  = eid,
					sE   = getSE(eid),
					part = part,
					obj  = model,
				}
			end
		end
		return out
	end

	local function scanTargetable()
		local world = getWorld()
		if not world then return {} end
		local out = {}
		if Components.Targetable and Components.ModelInstance then
			for eid, tg in world:query(Components.Targetable) do
				local mi = world:get(eid, Components.ModelInstance)
				local hp = Components.Health and world:get(eid, Components.Health)
				if mi and mi.model then
					local isMe = lp.Character and mi.model == lp.Character
					if not isMe then
						out[#out+1] = {
							eid      = eid,
							sE       = getSE(eid),
							model    = mi.model,
							hp       = hp,
							isPlayer = Components.PlayerInstance and world:get(eid, Components.PlayerInstance) ~= nil,
						}
					end
				end
			end
		end
		return out
	end

	local function scanZones()
		local world = getWorld()
		if not world then return {} end
		local out = {}
		if Components.Zone then
			for eid, zoneComp in world:query(Components.Zone) do
				local mi = Components.ModelInstance and world:get(eid, Components.ModelInstance)
				if mi and mi.model then
					out[#out+1] = { eid=eid, sE=getSE(eid), model=mi.model }
				end
			end
		end
		return out
	end

	local function scanCheckpoints()
		local world = getWorld()
		if not world then return {} end
		local out = {}
		if Components.Checkpoint then
			for eid, cpComp in world:query(Components.Checkpoint) do
				out[#out+1] = { eid=eid, sE=getSE(eid) }
			end
		end
		return out
	end

	local function scanWeapons()
		local world = getWorld()
		if not world then return {} end
		local entityToPlayer = {}
		if Components.ModelInstance then
			for eid, mi in world:query(Components.ModelInstance) do
				if mi.model then
					for _, p in ipairs(Players:GetPlayers()) do
						if p.Character and mi.model == p.Character then
							entityToPlayer[eid] = p.Name
							break
						end
					end
				end
			end
		end
		local out = {}
		if Components.ItemOwnedBy and Components.ModelInstance then
			for eid, iob in world:query(Components.ItemOwnedBy) do
				local desc = Components.ItemModelDescription and world:get(eid, Components.ItemModelDescription)
				local mi   = world:get(eid, Components.ModelInstance)
				if not iob or not iob.entity then continue end
				if not mi or not mi.model then continue end
				if mi.model.Name ~= "RightHandItem" then continue end
				local itemName = "?"
				if desc and desc.prefab then
					local ok, n = pcall(function() return desc.prefab.Name end)
					if ok then itemName = n end
				end
				if itemName == "?" then continue end
				local ownerName = entityToPlayer[iob.entity] or ("eid:"..tostring(iob.entity))
				out[#out+1] = { owner=ownerName, item=itemName, eid=eid }
			end
		end
		return out
	end

	--------------------------------------------------
	-- INTERACT / GRAB
	--------------------------------------------------
	local working = false

	local function grabItem(d, cb)
		if working or not hrp or not d.obj then cb(false) return end
		working = true
		local savedCF = hrp.CFrame
		hrp.CFrame = CFrame.new(itemPos(d) + Vector3.new(0, 3, 0))
		task.spawn(function()
			local got = false
			for i = 1, 50 do
				if not d.obj.Parent then got = true break end
				pcall(remInteract.FireServer, remInteract, d.sE)
				task.wait(0.03)
			end
			task.wait(0.05)
			if hrp and hrp.Parent then hrp.CFrame = savedCF end
			working = false
			cb(got)
		end)
	end

	local function interactEntity(sE, cb)
		if not sE then if cb then cb(false) end return end
		task.spawn(function()
			for i = 1, 5 do
				pcall(remInteract.FireServer, remInteract, sE)
				task.wait(0.05)
			end
			if cb then cb(true) end
		end)
	end

	local shardCollecting = false
	local function collectAllShards(statusPara)
		if shardCollecting or not hrp then return end
		shardCollecting = true
		task.spawn(function()
			local shards = scanShards()
			if #shards == 0 then
				statusPara:Set({Title="shards", Content="no shards found"})
				shardCollecting = false
				return
			end
			statusPara:Set({Title="shards", Content="collecting "..#shards.."..."})
			local savedCF = hrp.CFrame
			local collected = 0
			for _, d in ipairs(shards) do
				local pos = d.part and d.part.Parent and d.part.Position
				if pos then
					hrp.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
					task.wait(0.1)
				end
				local removed = false
				for i = 1, 20 do
					if d.part and not d.part.Parent then removed = true break end
					pcall(remInteract.FireServer, remInteract, d.sE)
					task.wait(0.04)
				end
				if removed or pos then collected += 1 end
				task.wait(0.05)
			end
			if hrp and hrp.Parent then hrp.CFrame = savedCF end
			shardCollecting = false
			statusPara:Set({Title="shards", Content="done  "..collected.."/"..#shards})
		end)
	end

	--------------------------------------------------
	local hoverActive      = false
	local hoverTarget      = nil
	local hoverOffsetY     = 15
	local hoverCamLock     = true
	local hoverConn        = nil
	local hoverAutoAttack  = false
	local hoverAttackDelay = 0.08
	local lastHoverAttack  = 0

	local function stopHover()
		hoverActive = false
		hoverTarget = nil
		if hoverConn then
			hoverConn:Disconnect()
			hoverConn = nil
		end
		Camera.CameraType = Enum.CameraType.Custom
	end

	local function startHover(target, offsetY, useCam)
		stopHover()
		if not target then return end

		hoverActive  = true
		hoverTarget  = target
		hoverOffsetY = offsetY or 15
		hoverCamLock = useCam ~= false

		hoverConn = RunService.Heartbeat:Connect(function()
			if not hoverActive then stopHover() return end

			local character = lp.Character
			if not character then stopHover() return end

			local model = hoverTarget.model
			if not model or not model.Parent then stopHover() return end

			local hb = getHitbox(model)
			if not hb or not hb.Parent then return end

			local abovePos = hb.Position + Vector3.new(0, hoverOffsetY, 0)

			pcall(function()
				character:PivotTo(CFrame.lookAt(abovePos, hb.Position))
			end)

			if hoverCamLock then
				Camera.CameraType = Enum.CameraType.Scriptable
				Camera.CFrame = CFrame.lookAt(abovePos + Vector3.new(0, 1.5, 0), hb.Position)
			end

			if hoverAutoAttack and remAction then
				local now = tick()
				if now - lastHoverAttack >= hoverAttackDelay then
					lastHoverAttack = now
					pcall(function()
						remAction:FireServer("usePrimary", Enum.UserInputState.Begin)
					end)
				end
			end
		end)
	end

	local hitboxActive = false

	local hitboxTransparency = 0.6
	local modifiedHitboxes = {}

	local function restoreHitboxes()
		for part, data in pairs(modifiedHitboxes) do
			if part and part.Parent then
				pcall(function()
					part.Size = data.origSize or Vector3.new(2, 2, 1)
					part.Transparency = data.origTransparency or 1
					part.CanCollide = data.origCanCollide or false
				end)
			end
		end
		table.clear(modifiedHitboxes)
	end

	local function updateHitboxes()
		if not hitboxActive then
			restoreHitboxes()
			return
		end
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= lp and p.Character then
				local hrp = p.Character:FindFirstChild("HumanoidRootPart") or p.Character:FindFirstChild("Torso")
				if hrp and hrp:IsA("BasePart") then
					if not modifiedHitboxes[hrp] then
						modifiedHitboxes[hrp] = {
							origSize = hrp.Size,
							origTransparency = hrp.Transparency,
							origCanCollide = hrp.CanCollide
						}
					end
					hrp.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
					hrp.Transparency = hitboxTransparency
					hrp.CanCollide = false
					hrp.Massless = true
				end
			end
		end
	end

	-- 2. Universal Auto-Parry State
	local autoParryActive = false

	local function triggerBlockAction(state)
		pcall(function()
			if remAction then
				remAction:FireServer("block", state)
			elseif remotes and remotes.player and remotes.player.input and remotes.player.input.action then
				remotes.player.input.action:fire("block", state)
			end
		end)
	end

	local function doParry(reason)
		local now = tick()
		if now - lastParryTick < (parryHoldDuration + 0.05) then return end
		lastParryTick = now
		
		task.spawn(function()
			triggerBlockAction(Enum.UserInputState.Begin)
			task.wait(parryHoldDuration)
			triggerBlockAction(Enum.UserInputState.End)
		end)
	end

	local knownAttackKeywords = {
		"attack", "m1", "m2", "swing", "slash", "stab", "strike", "thrust", "slam",
		"cleave", "smash", "ability", "combo", "stomp", "spear", "hammer", "greataxe",
		"dagger", "machete", "scythe", "hacksaw", "tomahawk", "crush", "punch", "kick",
		"root_trap", "toxic_gas", "spit", "bite", "claw", "reap", "charged", "dash_attack"
	}

	local function isAttackTrack(track)
		local name = (track.Name or ""):lower()
		for _, kw in ipairs(knownAttackKeywords) do
			if string.find(name, kw) then return true end
		end
		if track.Priority == Enum.AnimationPriority.Action or track.Priority == Enum.AnimationPriority.Action2 or track.Priority == Enum.AnimationPriority.Action3 or track.Priority == Enum.AnimationPriority.Action4 then
			if not string.find(name, "idle") and not string.find(name, "walk") and not string.find(name, "run") and not string.find(name, "jump") and not string.find(name, "fall") and not string.find(name, "sit") and not string.find(name, "hold") then
				return true
			end
		end
		return false
	end

	local function hookCharAnimations(c)
		if not c or c == lp.Character then return end
		local hum = c:FindFirstChildOfClass("Humanoid") or c:WaitForChild("Humanoid", 3)
		if not hum then return end
		local anim = hum:FindFirstChildOfClass("Animator") or hum:WaitForChild("Animator", 3)
		if not anim then return end

		anim.AnimationPlayed:Connect(function(track)
			if not autoParryActive then return end
			local myChar = lp.Character
			local myRoot = myChar and (myChar:FindFirstChild("HumanoidRootPart") or myChar:FindFirstChild("Torso"))
			local enemyRoot = c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("Torso") or c.PrimaryPart
			if not myRoot or not enemyRoot then return end

			local dist = (myRoot.Position - enemyRoot.Position).Magnitude
			if dist > parryRange then return end
			if not isAttackTrack(track) then return end

			local ping = 0.05
			pcall(function()
				ping = game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue() / 1000
			end)
			ping = math.clamp(ping, 0.02, 0.25)

			local animLen = track.Length > 0.1 and track.Length or 0.6
			local hitDelay = math.max(0.01, (animLen * 0.35) - ping)

			task.delay(hitDelay, function()
				if autoParryActive and c and c.Parent and myChar and myChar.Parent then
					local curDist = (myRoot.Position - enemyRoot.Position).Magnitude
					if curDist <= (parryRange + 6) then
						doParry("Attack: " .. tostring(track.Name))
					end
				end
			end)
		end)
	end

	for _, p in ipairs(Players:GetPlayers()) do

		p.CharacterAdded:Connect(hookCharAnimations)

	local function SetFreeCam(enable)
		if FC.Enabled == enable then return end

			FC.Roll = math.deg(rz)

			FC.RotVelocity = Vector2.zero
			FC.CurrentCFrame = Camera.CFrame

			if SharedRegistries and SharedRegistries.Cameras and SharedRegistries.Cameras.values then
				SharedRegistries.Cameras.values[NATIVE_CAM_ID] = nativeCameraDef
			end

				pcall(function() cameraManager:activateCamera(NATIVE_CAM_ID, {}) end)
			end

			Camera.CameraSubject = nil

			if FC.FrozenCharacter then
				if root then FC.OldRootAnchored = root.Anchored; root.Anchored = true end
				if hum then FC.OldWalkSpeed = hum.WalkSpeed; FC.OldJumpPower = hum.JumpPower; hum.WalkSpeed = 0; hum.JumpPower = 0 end
			end
			FC.MouseLookActive = true
			UserInputService.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition
		else
			if cameraManager then pcall(function() cameraManager:deactivateCamera(NATIVE_CAM_ID) end) end
			Camera.CameraType = FC.OldCameraType or Enum.CameraType.Custom
			if hum then Camera.CameraSubject = hum elseif root then Camera.CameraSubject = root end

		end
	end

	local function TeleportPlayerToCam()

		end
	end

	local lastFCTick = tick()
	local function stepFCSim()
		local now = tick()
		local dt = math.min(now - lastFCTick, 0.1)
		lastFCTick = now
		if not FC.Enabled then return end

		local isLooking = FC.HoldingRMB or FC.MouseLookActive
		if isLooking then
			local delta = UserInputService:GetMouseDelta()
			if delta.Magnitude > 0 then
				local targetRotVel = Vector2.new(-delta.Y * FC.MouseSensitivity, -delta.X * FC.MouseSensitivity)
				FC.RotVelocity = FC.Smoothness > 0 and FC.RotVelocity:Lerp(targetRotVel, math.clamp(1 - FC.Smoothness, 0.05, 1)) or targetRotVel
			else
				FC.RotVelocity = FC.RotVelocity:Lerp(Vector2.zero, 0.25)
			end
		else
			FC.RotVelocity = FC.RotVelocity:Lerp(Vector2.zero, 0.3)
		end

		FC.CamRot = FC.CamRot + FC.RotVelocity
		FC.CamRot = Vector2.new(math.clamp(FC.CamRot.X, -89.9, 89.9), FC.CamRot.Y % 360)
		local rotCFrame = CFrame.Angles(0, math.rad(FC.CamRot.Y), 0) * CFrame.Angles(math.rad(FC.CamRot.X), 0, math.rad(FC.Roll))

		local move = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + rotCFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - rotCFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - rotCFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + rotCFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) or UserInputService:IsKeyDown(Enum.KeyCode.E) then move = move + Vector3.new(0, 1, 0) end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.Q) then move = move - Vector3.new(0, 1, 0) end

		local curSpeed = FC.Speed
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then curSpeed = curSpeed * FC.BoostMultiplier
		elseif UserInputService:IsKeyDown(Enum.KeyCode.LeftAlt) then curSpeed = curSpeed * FC.SlowMultiplier end
		if move.Magnitude > 0 then move = move.Unit * curSpeed end

		FC.Velocity = FC.Smoothness > 0 and FC.Velocity:Lerp(move, math.clamp((1 - FC.Smoothness) * (dt * 60), 0.05, 1)) or move
		FC.CamPos = FC.CamPos + (FC.Velocity * dt)
		FC.CurrentCFrame = CFrame.new(FC.CamPos) * rotCFrame

		Camera.CameraType = Enum.CameraType.Scriptable
		Camera.CFrame = FC.CurrentCFrame
		Camera.Focus = FC.CurrentCFrame
		Camera.FieldOfView = FC.Fov
	end

		if hoverActive then
			task.wait(1)
			local t = hoverTarget
			if t then startHover(t, hoverOffsetY, hoverCamLock) end
		end
	end)

	local function shutdown()
		alive = false
		autoCollectEnabled = false
		stopHover()
		for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
		pcall(function() UI:Destroy() end)
		genv.__TDR_STOP = nil
	end
	genv.__TDR_STOP = shutdown

	conns[#conns+1] = UserInputService.InputBegan:Connect(function(inp, gp)
		if gp then return end
		if inp.KeyCode == Enum.KeyCode.RightShift then
			shutdown()
		end
	end)

	print("hub tdr loaded (velocity-compatible)")
end

task.spawn(sendLog)

if isMobile() then
	print("[Hub TDR] Platform: Mobile")
	runMobile()
else
	print("[Hub TDR] Platform: PC")
	runPC()
end
