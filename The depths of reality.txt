-- 1. MATTER ECS MODULE & REGISTRY INITIALIZATION

local RuntimeLib, GameWorld, Components, createItemFunc, SharedRegistries
local ecsReady = false

local function initECS()
    local ok = pcall(function()
        local rbxts = ReplicatedStorage:WaitForChild("rbxts_include", 5)
        if not rbxts then return end
        RuntimeLib = require(rbxts:WaitForChild("RuntimeLib", 5))
        
        GameWorld = RuntimeLib.import(script, ReplicatedStorage, "Game", "matter", "gameWorld").GameWorld
        Components = RuntimeLib.import(script, ReplicatedStorage, "Game", "matter", "components").Components
        createItemFunc = RuntimeLib.import(script, ReplicatedStorage, "Game", "matter", "utils", "createItem").createItem
        
        local regMod = ReplicatedStorage:FindFirstChild("Game") and ReplicatedStorage.Game:FindFirstChild("registries")
        if regMod then
            SharedRegistries = RuntimeLib.import(script, ReplicatedStorage, "Game", "registries").SharedRegistries
        end
        ecsReady = true
        print("[Hub TDR]: Matter ECS успешно инициализирован!")
    end)
    return ok and ecsReady
end

task.spawn(initECS)

-- 2. ECS QUERY WRAPPER & HELPER FUNCTIONS

local ECS = {}

function ECS.getWorld()
    if not ecsReady or not GameWorld then return nil end
    local ok, world = pcall(function() return GameWorld.getECSWorld() end)
    return ok and world or nil
end

function ECS.getLocalEntity()
    local world = ECS.getWorld()
    if not world or not GameWorld then return nil end
    local ok, entityId = pcall(function() return GameWorld.getLocalEntity(world) end)
    return ok and entityId or nil
end

function ECS.queryItems()
    local results = {}
    local world = ECS.getWorld()
    
    if world and Components and Components.Item and Components.ModelInstance then
        pcall(function()
            for entityId, itemComp, modelComp in world:query(Components.Item, Components.ModelInstance) do
                local model = modelComp and modelComp.model
                local itemData = itemComp and itemComp.item
                if model and model.Parent then
                    table.insert(results, {
                        entityId = entityId,
                        model = model,
                        item = itemData,
                        id = (itemData and itemData.id) or model.Name,
                        name = (itemData and itemData.id) or model.Name,
                        isECS = true
                    })
                end
            end
        end)
    end
    
    -- Дополнительный поиск в workspace для дропов
    if #results == 0 then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("Model") or obj:IsA("BasePart") then
                local name = obj.Name:lower()
                if name:find("item") or name:find("drop") or name:find("shard") or obj:FindFirstChild("Prompt") or obj:FindFirstChildOfClass("ProximityPrompt") then
                    table.insert(results, {
                        model = obj,
                        id = obj.Name,
                        name = obj.Name,
                        isECS = false
                    })
                end
            end
        end
    end
    
    return results
end

function ECS.queryShards()
    local allItems = ECS.queryItems()
    local shards = {}
    for _, it in ipairs(allItems) do
        local idLower = tostring(it.id):lower()
        if idLower:find("shard") or idLower:find("fragment") or idLower:find("crystal") or idLower:find("core") then
            table.insert(shards, it)
        end
    end
    return shards
end

function ECS.queryEntities()
    local results = {}
    local world = ECS.getWorld()
    local char = player.Character
    
    if world and Components and Components.ModelInstance then
        pcall(function()
            for entityId, modelComp in world:query(Components.ModelInstance) do
                local model = modelComp and modelComp.model
                if model and model.Parent and model ~= char then
                    local hum = model:FindFirstChildOfClass("Humanoid")
                    local healthComp = Components.Health and world:get(entityId, Components.Health)
                    local typeComp = Components.EntityTypeId and world:get(entityId, Components.EntityTypeId)
                    
                    if hum or healthComp or typeComp then
                        table.insert(results, {
                            entityId = entityId,
                            model = model,
                            name = (typeComp and typeComp.id) or model.Name,
                            health = (healthComp and healthComp.current) or (hum and hum.Health) or 0,
                            maxHealth = (healthComp and healthComp.max) or (hum and hum.MaxHealth) or 0,
                            isECS = true
                        })
                    end
                end
            end
        end)
    end
    
    if #results == 0 then
        for _, obj in ipairs(workspace:GetChildren()) do
            if obj:IsA("Model") and obj ~= char and obj:FindFirstChildOfClass("Humanoid") then
                local hum = obj:FindFirstChildOfClass("Humanoid")
                table.insert(results, {
                    model = obj,
                    name = obj.Name,
                    health = hum.Health,
                    maxHealth = hum.MaxHealth,
                    isECS = false
                })
            end
        end
    end
    
    return results
end

function ECS.dumpEntityComponents(entityId, model)
    local lines = {}
    local world = ECS.getWorld()
    
    if world and entityId and Components then
        table.insert(lines, "=== MATTER ECS ENTITY [" .. tostring(entityId) .. "] ===")
        for compName, compCtor in pairs(Components) do
            if type(compCtor) == "table" or type(compCtor) == "function" then
                local ok, data = pcall(function() return world:get(entityId, compCtor) end)
                if ok and data then
                    local jsonStr = ""
                    pcall(function() jsonStr = HttpService:JSONEncode(data) end)
                    table.insert(lines, "[Component] " .. tostring(compName) .. " -> " .. (jsonStr ~= "" and jsonStr or "Active"))
                end
            end
        end
    end
    
    if model then
        table.insert(lines, "=== ROBLOX INSTANCE HIERARCHY ===")
        for _, child in ipairs(model:GetDescendants()) do
            table.insert(lines, "[Instance] " .. child.Name .. " (" .. child.ClassName .. ")")
        end
    end
    
    return lines
end

function ECS.injectItems(itemIds)
    local world = ECS.getWorld()
    local localEntity = ECS.getLocalEntity()
    if not world or not localEntity or not createItemFunc or not Components then
        return false, "ECS не готов"
    end
    
    local ok, err = pcall(function()
        local inv = world:get(localEntity, Components.InventoryContainer)
        local hotbar = world:get(localEntity, Components.HotbarContainer)
        
        local newItems = {}
        for _, id in ipairs(itemIds) do
            table.insert(newItems, createItemFunc({ id = id, count = 1 }))
        end
        
        if inv then
            local items = table.clone(inv.items)
            for i, it in ipairs(newItems) do
                items[i] = it
            end
            world:insert(localEntity, inv:patch({ items = items }))
        end
        
        if hotbar then
            local hItems = table.clone(hotbar.items)
            for i, it in ipairs(newItems) do
                hItems[i] = it
            end
            world:insert(localEntity, hotbar:patch({ items = hItems }))
        end
    end)
    
    return ok, err
end

-- 9. KEYBINDS & MAIN HEARTBEAT LOOP

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.RightShift then
        isUIVisible = not isUIVisible
        mainFrame.Visible = isUIVisible
        if not isUIVisible then
            isCollecting = false
            isHovering = false
        end
    end
end)

local lastAttackTime = 0

RunService.Heartbeat:Connect(function()
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    -- Периодическое сканирование Matter ECS
    if tick() - lastRefresh > 1.2 then
        itemCache = ECS.queryItems()
        shardCache = ECS.queryShards()
        entityCache = ECS.queryEntities()
        lastRefresh = tick()
    end

    -- Обновление UI
    if isUIVisible then
        pcall(function()
            local worldReady = ECS.getWorld() ~= nil
            itemCountLabel.Text = string.format("Items in World: %d | Shards: %d", #itemCache, #shardCache)
            statusLabel.Text = string.format("Matter World: %s | LocalEntity: %s\nItems: %d | Shards: %d | Entities: %d\nAction: %s",
                worldReady and "CONNECTED (OK)" or "WAITING",
                tostring(ECS.getLocalEntity() or "N/A"),
                #itemCache,
                #shardCache,
                #entityCache,
                isCollecting and "Auto-Collecting" or (isHovering and "Hovering" or "Idle")
            )
        end)
    end

    -- Сбор предметов / осколков
    if isCollecting or collectShards then
        local targetList = collectShards and shardCache or itemCache
        for _, entry in ipairs(targetList) do
            local model = entry.model
            if model and model.Parent then
                pcall(function()
                    if model:IsA("Model") then
                        model:PivotTo(hrp.CFrame)
                    elseif model:IsA("BasePart") then
                        model.CFrame = hrp.CFrame
                    end
                end)
            end
        end
    end

    -- Парение над ближайшей сущностью
    if isHovering and #entityCache > 0 then
        local target = entityCache[1]
        local tModel = target and target.model
        if tModel and tModel.Parent then
            pcall(function()
                local tPos = tModel:GetPivot().Position
                hrp.CFrame = CFrame.new(tPos + Vector3.new(0, hoverOffset, 3), tPos)
                if cameraLocked then
                    camera.CFrame = CFrame.new(camera.CFrame.Position, tPos)
                end
            end)
        end
    end

    -- Авто-атака
    if autoAttack and #entityCache > 0 and (tick() - lastAttackTime > 0.4) then
        local target = entityCache[1]
        local tModel = target and target.model
        if tModel and tModel.Parent then
            local dist = (tModel:GetPivot().Position - hrp.Position).Magnitude
            if dist < 18 then
                mouse1click()
                lastAttackTime = tick()
            end
        end
    end
end)

print("[Hub TDR v2.0]: Скрипт успешно запущен. Нажмите RightShift для открытия меню.")
