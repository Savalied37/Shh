local function runMobile()
	local Players = game:GetService("Players")
	local Workspace = game:GetService("Workspace")
	local ReplicatedStorage = game:GetService("ReplicatedStorage")
	local RunService = game:GetService("RunService")
	local UserInputService = game:GetService("UserInputService")
	local Camera = Workspace.CurrentCamera

	local lp = Players.LocalPlayer

	--------------------------------------------------
	-- RAYFIELD
	--------------------------------------------------
	local Rayfield = loadstring(game:HttpGet("https://sirius.menu/rayfield"))()

	--------------------------------------------------
	-- ECS
	--------------------------------------------------
	local RL = require(ReplicatedStorage.rbxts_include.RuntimeLib)
	local GameWorld = RL.import(script, ReplicatedStorage, "Game", "matter", "gameWorld").GameWorld
	local Components = RL.import(script, ReplicatedStorage, "Game", "matter", "components").Components
	local findTopmost = RL.import(script, ReplicatedStorage, "Game", "matter", "utils", "interactable").findTopmostEntityAncestor

	local cnt = ReplicatedStorage.rbxts_include.node_modules["@rbxts"].remo.src.container
	local remInteract = cnt["player.input.interact"]

	local remAction = nil
	pcall(function()
		remAction = cnt["player.input.action"]
	end)

	local char, hrp
	local function bindChar(c)
		char = c
		hrp  = c:WaitForChild("HumanoidRootPart")
	end
	if lp.Character then bindChar(lp.Character) end
	lp.CharacterAdded:Connect(bindChar)

	--------------------------------------------------
	-- ECS HELPERS
	--------------------------------------------------
	local function getWorld() return GameWorld.getECSWorld() end

	local function getSE(eid)
		local ok, sid = pcall(GameWorld.getServerId, eid)
		return ok and sid or nil
	end

	local function getHitbox(model)
		if not model then return nil end
		if model:IsA("BasePart") then return model end
		if model:IsA("Model") then
			local h = model:FindFirstChild("HumanoidRootPart")
			if h then return h end
			if model.PrimaryPart then return model.PrimaryPart end
			for _, p in ipairs(model:GetDescendants()) do
				if p:IsA("BasePart") then return p end
			end
		end
		return nil
	end

	local function getModelFromEid(eid)
		local world = getWorld()
		local mi = world:get(eid, Components.ModelInstance)
		if mi and mi.model then return mi.model end
		local tr = world:get(eid, Components.Tracer)
		if tr and tr.instance then
			local inst = tr.instance
			if inst:IsA("Model") then return inst end
			return inst:FindFirstAncestorOfClass("Model")
		end
		return nil
	end

	local function entityPos(eid)
		local m = getModelFromEid(eid)
		if not m then return nil end
		local hb = getHitbox(m)
		return hb and hb.Position or nil
	end

	local function tpToModel(model, offsetY)
		if not hrp or not model then return end
		local hb = getHitbox(model)
		if hb then hrp.CFrame = CFrame.new(hb.Position + Vector3.new(0, offsetY or 0, 0)) end
	end

	--------------------------------------------------
	-- CAMERA
	--------------------------------------------------
	local function stopCamLock()
		Camera.CameraType = Enum.CameraType.Custom
	end

	--------------------------------------------------
	-- SHARD EID SET
	--------------------------------------------------
	local shardEids = {}

	local function buildShardEidSet()
		local world = getWorld()
		shardEids = {}
		for eid in world do
			if world:get(eid, Components.ShardSpawner) then shardEids[eid] = true end
		end
	end

	--------------------------------------------------
	-- SCANNERS
	--------------------------------------------------
	local itemCache, itemStamp = {}, 0
	local function scanItems()
		if tick() - itemStamp < 2 then return itemCache end
		itemStamp = tick()
		buildShardEidSet()
		local world = getWorld()
		local out, seen = {}, {}
		for _, part in ipairs(Workspace:GetDescendants()) do
			if part:IsA("BasePart") and part.CollisionGroup == "Interactable" then
				local ok, res = pcall(findTopmost, part)
				if ok and res and res.model and not seen[res.model] then
					local eid = res.entity
					if shardEids[eid] then continue end
					if world:get(eid, Components.Interactable) and world:get(eid, Components.Item) then
						seen[res.model] = true
						out[#out+1] = {
							eid  = eid,
							sE   = GameWorld.getServerId(eid),
							obj  = res.model,
							part = part,
						}
					end
				end
			end
		end
		itemCache = out
		return out
	end

	local function itemPos(d)
		if d.part and d.part:IsA("BasePart") then return d.part.Position end
		local m = d.obj
		if m:IsA("Model") then
			local ok, pv = pcall(m.GetPivot, m)
			if ok then return pv.Position end
			if m.PrimaryPart then return m.PrimaryPart.Position end
		end
		return Vector3.zero
	end

	local function scanShards()
		local world = getWorld()
		local shardList = {}
		for eid in world do
			if world:get(eid, Components.ShardSpawner) and world:get(eid, Components.Interactable) then
				shardList[#shardList+1] = { eid=eid, sE=getSE(eid) }
			end
		end
		local eidToPart = {}
		for _, part in ipairs(Workspace:GetDescendants()) do
			if part:IsA("BasePart") and part.CollisionGroup == "Interactable" then
				local ok, res = pcall(findTopmost, part)
				if ok and res and not eidToPart[res.entity] then
					eidToPart[res.entity] = part
				end
			end
		end
		local out = {}
		for _, d in ipairs(shardList) do
			out[#out+1] = { eid=d.eid, sE=d.sE, part=eidToPart[d.eid] }
		end
		return out
	end

	local function scanTargetable()
		local world = getWorld()
		local out = {}
		for eid in world do
			local tg = world:get(eid, Components.Targetable)
			local mi = world:get(eid, Components.ModelInstance)
			local hp = world:get(eid, Components.Health)
			if tg and mi then
				local isMe = lp.Character and mi.model == lp.Character
				if not isMe then
					out[#out+1] = {
						eid      = eid,
						sE       = getSE(eid),
						model    = mi.model,
						hp       = hp,
						isPlayer = world:get(eid, Components.PlayerInstance) ~= nil,
					}
				end
			end
		end
		return out
	end

	local function scanZones()
		local world = getWorld()
		local out = {}
		for eid in world do
			if world:get(eid, Components.Zone) then
				local mi = world:get(eid, Components.ModelInstance)
				if mi then out[#out+1] = { eid=eid, sE=getSE(eid), model=mi.model } end
			end
		end
		return out
	end

	local function scanCheckpoints()
		local world = getWorld()
		local out = {}
		for eid in world do
			if world:get(eid, Components.Checkpoint) and world:get(eid, Components.Interactable) then
				out[#out+1] = { eid=eid, sE=getSE(eid) }
			end
		end
		return out
	end

	local function scanWeapons()
		local world = getWorld()
		local entityToPlayer = {}
		for eid in world do
			local mi = world:get(eid, Components.ModelInstance)
			local hi = world:get(eid, Components.HumanoidInstance)
			local pi = world:get(eid, Components.PlayerInstance)
			if (hi or pi) and mi then
				for _, p in ipairs(Players:GetPlayers()) do
					if p.Character and mi.model == p.Character then
						entityToPlayer[eid] = p.Name
						break
					end
				end
			end
		end
		local out = {}
		for eid in world do
			local iob  = world:get(eid, Components.ItemOwnedBy)
			local desc = world:get(eid, Components.ItemModelDescription)
			local mi   = world:get(eid, Components.ModelInstance)
			if not iob or not iob.entity then continue end
			if not mi or not mi.model then continue end
			if mi.model.Name ~= "RightHandItem" then continue end
			local itemName = "?"
			if desc and desc.prefab then
				local ok, n = pcall(function() return desc.prefab.Name end)
				if ok then itemName = n end
			end
			if itemName == "?" then continue end
			local ownerName = entityToPlayer[iob.entity] or ("eid:"..tostring(iob.entity))
			out[#out+1] = { owner=ownerName, item=itemName, eid=eid }
		end
		return out
	end

	--------------------------------------------------
	-- INTERACT / GRAB
	--------------------------------------------------
	local working = false

	local function grabItem(d, cb)
		if working or not hrp then cb(false) return end
		working = true
		local savedCF = hrp.CFrame
		hrp.CFrame = CFrame.new(itemPos(d) + Vector3.new(0, 3, 0))
		task.spawn(function()
			local got = false
			for i = 1, 50 do
				if not d.obj.Parent then got=true break end
				pcall(remInteract.FireServer, remInteract, d.sE)
				task.wait(0.03)
			end
			task.wait(0.05)
			if hrp and hrp.Parent then hrp.CFrame = savedCF end
			working = false
			cb(got)
		end)
	end

	local function interactEntity(sE, cb)
		if not sE then if cb then cb(false) end return end
		task.spawn(function()
			for i = 1, 5 do
				pcall(remInteract.FireServer, remInteract, sE)
				task.wait(0.05)
			end
			if cb then cb(true) end
		end)
	end

	local shardCollecting = false
	local function collectAllShards(statusPara)
		if shardCollecting or not hrp then return end
		shardCollecting = true
		task.spawn(function()
			local shards = scanShards()
			if #shards == 0 then
				statusPara:Set({Title="shards", Content="no shards found"})
				shardCollecting = false
				return
			end
			statusPara:Set({Title="shards", Content="collecting "..#shards.."..."})
			local savedCF = hrp.CFrame
			local collected = 0
			for _, d in ipairs(shards) do
				local pos = d.part and d.part.Parent and d.part.Position
				if pos then
					hrp.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
					task.wait(0.1)
				end
				local removed = false
				for i = 1, 20 do
					if d.part and not d.part.Parent then removed=true break end
					pcall(remInteract.FireServer, remInteract, d.sE)
					task.wait(0.04)
				end
				if removed or pos then collected+=1 end
				task.wait(0.05)
			end
			if hrp and hrp.Parent then hrp.CFrame = savedCF end
			shardCollecting = false
			statusPara:Set({Title="shards", Content="done  "..collected.."/"..#shards})
		end)
	end

	--------------------------------------------------
	-- HOVER SYSTEM
	--------------------------------------------------
	local hoverActive     = false
	local hoverTarget     = nil
	local hoverOffsetY    = 15
	local hoverCamLock    = true
	local hoverConn       = nil
	local hoverAutoAttack = false
	local hoverAttackDelay= 0.08
	local lastHoverAttack = 0

	local function stopHover()
		hoverActive = false
		hoverTarget = nil
		if hoverConn then
			hoverConn:Disconnect()
			hoverConn = nil
		end
		Camera.CameraType = Enum.CameraType.Custom
	end

	local function startHover(target, offsetY, useCam)
		stopHover()
		if not target then return end

		hoverActive  = true
		hoverTarget  = target
		hoverOffsetY = offsetY or 15
		hoverCamLock = useCam ~= false

		hoverConn = RunService.Heartbeat:Connect(function()
			if not hoverActive then
				stopHover()
				return
			end

			local character = lp.Character
			if not character then
				stopHover()
				return
			end

			local model = hoverTarget.model
			if not model or not model.Parent then
				stopHover()
				return
			end

			local hb = getHitbox(model)
			if not hb or not hb.Parent then return end

			-- hoverOffsetY can be positive (above) or negative (below)
			local abovePos = hb.Position + Vector3.new(0, hoverOffsetY, 0)

			pcall(function()
				character:PivotTo(CFrame.lookAt(abovePos, hb.Position))
			end)

			if hoverCamLock then
				Camera.CameraType = Enum.CameraType.Scriptable
				Camera.CFrame = CFrame.lookAt(
					abovePos + Vector3.new(0, 1.5, 0),
					hb.Position
				)
			end

			if hoverAutoAttack and remAction then
				local now = tick()
				if now - lastHoverAttack >= hoverAttackDelay then
					lastHoverAttack = now
					pcall(function()
						remAction:FireServer("usePrimary", Enum.UserInputState.Begin)
					end)
				end
			end
		end)
	end

	TabItems:CreateButton({Name="tp to searched", Callback=function()

		for _, d in ipairs(items) do
			if d.obj.Name:lower():find(searchStr, 1, true) then found = d break end

		if not found then
			pItemStatus:Set({Title="info", Content="not found: "..searchStr})
			return

		pItemStatus:Set({Title="info", Content="going → "..found.obj.Name})

			pItemStatus:Set({Title="info", Content=(ok and "got " or "failed ")..found.obj.Name})

	TabItems:CreateButton({Name="tp to all items", Callback=function()
		local items = scanItems()
		if #items == 0 then
			pItemStatus:Set({Title="info", Content="nothing"})
			return
		end
		local n = 0
		task.spawn(function()
			for _, d in ipairs(items) do
				if not d.obj.Parent then continue end
				local done = false
				grabItem(d, function(ok)
					if ok then n += 1 end
					done = true
				end)
				while not done do task.wait(0.05) end
				task.wait(0.1)
			end
			pItemStatus:Set({Title="info", Content="done "..n.."/"..#items})
		end)

	TabItems:CreateButton({Name="show list", Callback=function()
		itemStamp = 0
		local items = scanItems()
		if #items == 0 then
			pItemList:Set({Title="list (0)", Content="empty"})
			return
		end
		local lines = {}
		for i, d in ipairs(items) do
			local dist = hrp and math.floor((itemPos(d) - hrp.Position).Magnitude) or 0
			lines[i] = ("[%d] %s  %dm"):format(i, d.obj.Name, dist)
			if i >= 25 then
				lines[i+1] = "... +"..(#items-25).." more"
				break
			end
		end
		pItemList:Set({Title="list ("..#items..")", Content=table.concat(lines, "\n")})

		Name="index", PlaceholderText="number...",

	TabItems:CreateButton({Name="tp by index", Callback=function()

		if not d then

	local function imp(...)
		local m = ReplicatedStorage
		for _, n in ipairs({ ... }) do
			m = m:WaitForChild(n, 10)
			if not m then error("[Hub TDR] Module not found: " .. tostring(n)) end
		end
		local ok, res, done
		task.spawn(function()
			ok, res = pcall(require, m)
			done = true
		end)
		repeat task.wait() until done
		if not ok then error(res) end
		return res
	end

	local function getHitbox(model)
		if not model then return nil end
		if model:IsA("BasePart") then return model end
		if model:IsA("Model") then
			local h = model:FindFirstChild("HumanoidRootPart")
			if h then return h end
			if model.PrimaryPart then return model.PrimaryPart end
			for _, p in ipairs(model:GetDescendants()) do
				if p:IsA("BasePart") then return p end
			end
		end
		return nil
	end

	local function getModelFromEid(eid)
		local world = getWorld()
		local mi = world:get(eid, Components.ModelInstance)
		if mi and mi.model then return mi.model end
		local tr = world:get(eid, Components.Tracer)
		if tr and tr.instance then
			local inst = tr.instance
			if inst:IsA("Model") then return inst end
			return inst:FindFirstAncestorOfClass("Model")
		end
		return nil
	end

	local function entityPos(eid)
		local m = getModelFromEid(eid)
		if not m then return nil end
		local hb = getHitbox(m)
		return hb and hb.Position or nil
	end

	local function tpToModel(model, offsetY)
		if not hrp or not model then return end
		local hb = getHitbox(model)
		if hb then hrp.CFrame = CFrame.new(hb.Position + Vector3.new(0, offsetY or 0, 0)) end
	end

	local function stopCamLock()
		Camera.CameraType = Enum.CameraType.Custom
	end

	--------------------------------------------------
	-- SHARD EID SET
	--------------------------------------------------
	local shardEids = {}

	local function buildShardEidSet()
		local world = getWorld()
		shardEids = {}
		for eid in world do
			if world:get(eid, Components.ShardSpawner) then shardEids[eid] = true end
		end
	end

	--------------------------------------------------
	-- SCANNERS
	--------------------------------------------------
	local itemCache, itemStamp = {}, 0
	local function scanItems()
		if tick() - itemStamp < 2 then return itemCache end
		itemStamp = tick()
		buildShardEidSet()
		local world = getWorld()
		local out, seen = {}, {}
		for _, part in ipairs(Workspace:GetDescendants()) do
			if part:IsA("BasePart") and part.CollisionGroup == "Interactable" then
				local ok, res = pcall(findTopmost, part)
				if ok and res and res.model and not seen[res.model] then
					local eid = res.entity
					if shardEids[eid] then continue end
					if world:get(eid, Components.Interactable) and world:get(eid, Components.Item) then
						seen[res.model] = true
						out[#out+1] = {
							eid  = eid,
							sE   = GameWorld.getServerId(eid),
							obj  = res.model,
							part = part,
						}
					end
				end
			end
		end
		itemCache = out
		return out
	end

	local function itemPos(d)
		if d.part and d.part:IsA("BasePart") then return d.part.Position end
		local m = d.obj
		if m:IsA("Model") then
			local ok, pv = pcall(m.GetPivot, m)
			if ok then return pv.Position end
			if m.PrimaryPart then return m.PrimaryPart.Position end
		end
		return Vector3.zero
	end

	local function scanShards()
		local world = getWorld()
		local shardList = {}
		for eid in world do
			if world:get(eid, Components.ShardSpawner) and world:get(eid, Components.Interactable) then
				shardList[#shardList+1] = { eid=eid, sE=getSE(eid) }
			end
		end
		local eidToPart = {}
		for _, part in ipairs(Workspace:GetDescendants()) do
			if part:IsA("BasePart") and part.CollisionGroup == "Interactable" then
				local ok, res = pcall(findTopmost, part)
				if ok and res and not eidToPart[res.entity] then
					eidToPart[res.entity] = part
				end
			end
		end
		local out = {}
		for _, d in ipairs(shardList) do
			out[#out+1] = { eid=d.eid, sE=d.sE, part=eidToPart[d.eid] }
		end
		return out
	end

	local function scanTargetable()
		local world = getWorld()
		local out = {}
		for eid in world do
			local tg = world:get(eid, Components.Targetable)
			local mi = world:get(eid, Components.ModelInstance)
			local hp = world:get(eid, Components.Health)
			if tg and mi then
				local isMe = lp.Character and mi.model == lp.Character
				if not isMe then
					out[#out+1] = {
						eid      = eid,
						sE       = getSE(eid),
						model    = mi.model,
						hp       = hp,
						isPlayer = world:get(eid, Components.PlayerInstance) ~= nil,
					}
				end
			end
		end
		return out
	end

	local function scanZones()
		local world = getWorld()
		local out = {}
		for eid in world do
			if world:get(eid, Components.Zone) then
				local mi = world:get(eid, Components.ModelInstance)
				if mi then out[#out+1] = { eid=eid, sE=getSE(eid), model=mi.model } end
			end
		end
		return out
	end

	local function scanCheckpoints()
		local world = getWorld()
		local out = {}
		for eid in world do
			if world:get(eid, Components.Checkpoint) and world:get(eid, Components.Interactable) then
				out[#out+1] = { eid=eid, sE=getSE(eid) }
			end
		end
		return out
	end

	local function scanWeapons()
		local world = getWorld()
		local entityToPlayer = {}
		for eid in world do
			local mi = world:get(eid, Components.ModelInstance)
			local hi = world:get(eid, Components.HumanoidInstance)
			local pi = world:get(eid, Components.PlayerInstance)
			if (hi or pi) and mi then
				for _, p in ipairs(Players:GetPlayers()) do
					if p.Character and mi.model == p.Character then
						entityToPlayer[eid] = p.Name
						break
					end
				end
			end
		end
		local out = {}
		for eid in world do
			local iob  = world:get(eid, Components.ItemOwnedBy)
			local desc = world:get(eid, Components.ItemModelDescription)
			local mi   = world:get(eid, Components.ModelInstance)
			if not iob or not iob.entity then continue end
			if not mi or not mi.model then continue end
			if mi.model.Name ~= "RightHandItem" then continue end
			local itemName = "?"
			if desc and desc.prefab then
				local ok, n = pcall(function() return desc.prefab.Name end)
				if ok then itemName = n end
			end
			if itemName == "?" then continue end
			local ownerName = entityToPlayer[iob.entity] or ("eid:"..tostring(iob.entity))
			out[#out+1] = { owner=ownerName, item=itemName, eid=eid }
		end
		return out
	end

	--------------------------------------------------
	-- INTERACT / GRAB
	--------------------------------------------------
	local working = false

	local function grabItem(d, cb)
		if working or not hrp then cb(false) return end
		working = true
		local savedCF = hrp.CFrame
		hrp.CFrame = CFrame.new(itemPos(d) + Vector3.new(0, 3, 0))
		task.spawn(function()
			local got = false
			for i = 1, 50 do
				if not d.obj.Parent then got = true break end
				pcall(remInteract.FireServer, remInteract, d.sE)
				task.wait(0.03)
			end
			task.wait(0.05)
			if hrp and hrp.Parent then hrp.CFrame = savedCF end
			working = false
			cb(got)
		end)
	end

	local function interactEntity(sE, cb)
		if not sE then if cb then cb(false) end return end
		task.spawn(function()
			for i = 1, 5 do
				pcall(remInteract.FireServer, remInteract, sE)
				task.wait(0.05)
			end
			if cb then cb(true) end
		end)
	end

	local shardCollecting = false
	local function collectAllShards(statusPara)
		if shardCollecting or not hrp then return end
		shardCollecting = true
		task.spawn(function()
			local shards = scanShards()
			if #shards == 0 then
				statusPara:Set({Title="shards", Content="no shards found"})
				shardCollecting = false
				return
			end
			statusPara:Set({Title="shards", Content="collecting "..#shards.."..."})
			local savedCF = hrp.CFrame
			local collected = 0
			for _, d in ipairs(shards) do
				local pos = d.part and d.part.Parent and d.part.Position
				if pos then
					hrp.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
					task.wait(0.1)
				end
				local removed = false
				for i = 1, 20 do
					if d.part and not d.part.Parent then removed = true break end
					pcall(remInteract.FireServer, remInteract, d.sE)
					task.wait(0.04)
				end
				if removed or pos then collected += 1 end
				task.wait(0.05)
			end
			if hrp and hrp.Parent then hrp.CFrame = savedCF end
			shardCollecting = false
			statusPara:Set({Title="shards", Content="done  "..collected.."/"..#shards})
		end)
	end

	--------------------------------------------------
	-- HOVER SYSTEM
	--------------------------------------------------
	local hoverActive      = false
	local hoverTarget      = nil
	local hoverOffsetY     = 15
	local hoverCamLock     = true
	local hoverConn        = nil
	local hoverAutoAttack  = false
	local hoverAttackDelay = 0.08
	local lastHoverAttack  = 0

	local function stopHover()
		hoverActive = false
		hoverTarget = nil
		if hoverConn then
			hoverConn:Disconnect()
			hoverConn = nil
		end
		Camera.CameraType = Enum.CameraType.Custom
	end

	local function startHover(target, offsetY, useCam)
		stopHover()
		if not target then return end

		hoverActive  = true
		hoverTarget  = target
		hoverOffsetY = offsetY or 15
		hoverCamLock = useCam ~= false

		hoverConn = RunService.Heartbeat:Connect(function()
			if not hoverActive then stopHover() return end

			local character = lp.Character
			if not character then stopHover() return end

			local model = hoverTarget.model
			if not model or not model.Parent then stopHover() return end

			local hb = getHitbox(model)
			if not hb or not hb.Parent then return end

			-- hoverOffsetY can be positive (above) or negative (below)
			local abovePos = hb.Position + Vector3.new(0, hoverOffsetY, 0)

			pcall(function()
				character:PivotTo(CFrame.lookAt(abovePos, hb.Position))
			end)

			if hoverCamLock then
				Camera.CameraType = Enum.CameraType.Scriptable
				Camera.CFrame = CFrame.lookAt(abovePos + Vector3.new(0, 1.5, 0), hb.Position)
			end

			if hoverAutoAttack and remAction then
				local now = tick()
				if now - lastHoverAttack >= hoverAttackDelay then
					lastHoverAttack = now
					pcall(function()
						remAction:FireServer("usePrimary", Enum.UserInputState.Begin)
					end)
				end
			end
		end)
	end

	end})

	TabMisc:CreateParagraph({
		Title   = "info",
		Content = "RightShift — stop hover + close ui\nScript execution is logged (username, game, place, HWID, time)."
	})

	--------------------------------------------------
	-- LOOPS / CLEANUP
	--------------------------------------------------
	local alive = true
	local conns = {}

	task.spawn(function()
		while alive do
			task.wait(0.5)
			if hoverActive and hoverTarget then
				local aaStr = hoverAutoAttack and "  ⚔ auto" or ""
				pHoverStatus:Set({
					Title   = "hover ✓",
					Content = "eid:"..tostring(hoverTarget.eid).."  Y:"..hoverOffsetY..aaStr
				})
			end
		end
	end)

	task.spawn(function()
		while alive do
			task.wait(5)
			itemStamp = 0
			pcall(scanItems)
		end
	end)

	conns[#conns+1] = lp.CharacterAdded:Connect(function(c)
		bindChar(c)
		if hoverActive then
			task.wait(1)
			local t = hoverTarget
			if t then startHover(t, hoverOffsetY, hoverCamLock) end
		end
	end)

	local function shutdown()
		alive = false
		stopHover()
		for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
		pcall(function() UI:Destroy() end)
		genv.__TDR_STOP = nil
	end
	genv.__TDR_STOP = shutdown

	conns[#conns+1] = UserInputService.InputBegan:Connect(function(inp, gp)
		if gp then return end
		if inp.KeyCode == Enum.KeyCode.RightShift then
			shutdown()
		end
	end)

	print("hub tdr loaded (velocity-compatible)")
end

--------------------------------------------------
-- INITIALIZATION
--------------------------------------------------
task.spawn(sendLog)

if isMobile() then
	print("[Hub TDR] Platform: Mobile")
	runMobile()
else
	print("[Hub TDR] Platform: PC")
	runPC()
end
